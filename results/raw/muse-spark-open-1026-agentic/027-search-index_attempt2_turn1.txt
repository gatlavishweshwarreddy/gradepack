class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._ngram_index = {}

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        # Build n-gram inverted index for 1,2,3-grams to support
        # O(1) substring search via posting list intersection.
        self._ngram_index = {}
        for doc_id, text in self._docs.items():
            L = len(text)
            # index all 1-grams, 2-grams and 3-grams
            for n in (1, 2, 3):
                if L < n:
                    continue
                for i in range(L - n + 1):
                    ngram = text[i:i+n]
                    bucket = self._ngram_index.get(ngram)
                    if bucket is None:
                        self._ngram_index[ngram] = {doc_id}
                    else:
                        bucket.add(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query using inverted index."""
        q = query.lower()
        if q == "":
            # "" in text is True for all docs in Python
            return list(self._docs.keys())

        if not self._built:
            self.build()

        # Short queries (<=3) are directly indexed: the posting list is exact
        if len(q) <= 3:
            candidates = self._ngram_index.get(q)
            if not candidates:
                return []
            # return in insertion order to match naive scan order
            return [doc_id for doc_id in self._docs if doc_id in candidates]

        # Long queries: intersect trigram posting lists to get candidates
        trigrams = [q[i:i+3] for i in range(len(q) - 2)]
        posting_lists = []
        for tri in trigrams:
            lst = self._ngram_index.get(tri)
            if lst is None:
                return []
            posting_lists.append(lst)

        # intersect starting with smallest list
        posting_lists.sort(key=len)
        candidates = posting_lists[0].copy()
        for lst in posting_lists[1:]:
            candidates.intersection_update(lst)
            if not candidates:
                return []

        # Verify candidates with actual substring check (only on candidates)
        results = []
        for doc_id in self._docs:
            if doc_id not in candidates:
                continue
            self.comparison_count += 1
            if q in self._docs[doc_id]:
                results.append(doc_id)
        return results
