import threading

_cache = {}
_lock = threading.RLock()
_compute_locks = {}
_compute_locks_lock = threading.Lock()

def memoize(fn):
    def wrapper(arg):
        # Fast path: check cache with lock
        with _lock:
            if arg in _cache:
                return _cache[arg]

        # Need to compute - ensure only one thread computes per key
        # Get or create per-key lock
        with _compute_locks_lock:
            if arg not in _compute_locks:
                _compute_locks[arg] = threading.Lock()
            key_lock = _compute_locks[arg]

        with key_lock:
            # Double-check cache after acquiring key lock
            with _lock:
                if arg in _cache:
                    return _cache[arg]

            # Compute outside global lock to avoid blocking other keys
            result = fn(arg)

            with _lock:
                _cache[arg] = result
            return result
    return wrapper

def clear_cache():
    with _lock:
        _cache.clear()
    with _compute_locks_lock:
        _compute_locks.clear()
