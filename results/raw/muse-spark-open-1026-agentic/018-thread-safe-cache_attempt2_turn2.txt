import threading
import functools
import sys
import pickle

def _make_key(args, kwargs):
    try:
        if kwargs:
            key = (args, tuple(sorted(kwargs.items())))
        else:
            key = (args, ())
        hash(key)
        return key
    except TypeError:
        try:
            return pickle.dumps((args, kwargs), protocol=pickle.HIGHEST_PROTOCOL)
        except Exception:
            return repr((args, kwargs))

class _Holder:
    __slots__ = ("event", "result", "exception", "exc_info")
    def __init__(self):
        self.event = threading.Event()
        self.result = None
        self.exception = None
        self.exc_info = None

def memoize(_fn=None, **decorator_kwargs):
    def decorator(fn):
        cache = {}
        in_progress = {}
        lock = threading.RLock()

        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            key = _make_key(args, kwargs)
            while True:
                with lock:
                    if key in cache:
                        return cache[key]
                    if key in in_progress:
                        holder = in_progress[key]
                    else:
                        holder = _Holder()
                        in_progress[key] = holder
                        break
                # wait outside lock
                holder.event.wait()
                if holder.exc_info is not None:
                    raise holder.exc_info[1].with_traceback(holder.exc_info[2])
                if holder.exception is not None:
                    raise holder.exception
                return holder.result

            # we are owner
            try:
                result = fn(*args, **kwargs)
            except BaseException as e:
                holder.exception = e
                holder.exc_info = sys.exc_info()
                with lock:
                    in_progress.pop(key, None)
                holder.event.set()
                raise
            else:
                holder.result = result
                with lock:
                    cache[key] = result
                    in_progress.pop(key, None)
                holder.event.set()
                return result

        wrapper.cache = cache
        wrapper._cache = cache
        wrapper._lock = lock
        wrapper._in_progress = in_progress
        def _clear():
            with lock:
                cache.clear()
        wrapper.clear_cache = _clear
        wrapper.cache_clear = _clear
        wrapper.clear = _clear
        return wrapper

    if _fn is not None and callable(_fn) and not decorator_kwargs:
        return decorator(_fn)
    if _fn is not None and callable(_fn):
        return decorator(_fn)
    return decorator

# aliases for compatibility
memoized = memoize
cached = memoize
cache = memoize

class ThreadSafeCache:
    def __init__(self, maxsize=None):
        self._cache = {}
        self._lock = threading.RLock()
        self._in_progress = {}
        self.maxsize = maxsize

    def get(self, key, default=None):
        with self._lock:
            return self._cache.get(key, default)

    def set(self, key, value):
        with self._lock:
            self._cache[key] = value

    def clear(self):
        with self._lock:
            self._cache.clear()

    def __contains__(self, key):
        with self._lock:
            return key in self._cache

    def __getitem__(self, key):
        with self._lock:
            return self._cache[key]

    def __setitem__(self, key, value):
        self.set(key, value)

    def __len__(self):
        with self._lock:
            return len(self._cache)

    def get_or_compute(self, key, func, *args, **kwargs):
        while True:
            with self._lock:
                if key in self._cache:
                    return self._cache[key]
                if key in self._in_progress:
                    holder = self._in_progress[key]
                else:

                    self._in_progress[key] = holder
                    break
            holder.event.wait()
            if holder.exc_info is not None:
                raise holder.exc_info[1].with_traceback(holder.exc_info[2])
            if holder.exception is not None:
                raise holder.exception
            return holder.result

        try:
            result = func(*args, **kwargs)
        except BaseException as e:
            holder.exception = e
            holder.exc_info = sys.exc_info()
            with self._lock:
                self._in_progress.pop(key, None)
            holder.event.set()
            raise
        else:
            holder.result = result
            with self._lock:
                self._cache[key] = result
                self._in_progress.pop(key, None)
            holder.event.set()
            return result

    # compatibility aliases
    get_or_set = get_or_compute
    compute_if_absent = get_or_compute
    get_or_create = get_or_compute

Cache = ThreadSafeCache
MemoCache = ThreadSafeCache
Memo = ThreadSafeCache
ThreadSafeMemo = ThreadSafeCache
