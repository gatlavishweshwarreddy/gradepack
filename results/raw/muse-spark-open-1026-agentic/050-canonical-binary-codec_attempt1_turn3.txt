import struct

TAG_NULL = 0x00
TAG_FALSE = 0x01
TAG_TRUE = 0x02
TAG_INT = 0x03
TAG_FLOAT = 0x04
TAG_STR = 0x05
TAG_LIST = 0x06
TAG_DICT = 0x07
TAG_SET = 0x08
TAG_REF = 0x09
TAG_TUPLE = 0x0A
TAG_BYTES = 0x0B


def dumps(obj):
    memo = {}
    out = bytearray()

    def write_varint(n):
        while n >= 0x80:
            out.append((n & 0x7F) | 0x80)
            n >>= 7
        out.append(n)

    def encode(o):
        oid = id(o)
        if isinstance(o, (list, dict, set, tuple)):
            if oid in memo:
                out.append(TAG_REF)
                write_varint(memo[oid])
                return
            memo[oid] = len(memo)
            if isinstance(o, list):
                out.append(TAG_LIST)
                write_varint(len(o))
                for item in o:
                    encode(item)
                return
            elif isinstance(o, tuple):
                out.append(TAG_TUPLE)
                write_varint(len(o))
                for item in o:
                    encode(item)
                return
            elif isinstance(o, dict):
                out.append(TAG_DICT)
                write_varint(len(o))
                try:
                    sorted_keys = sorted(o.keys())
                except TypeError:
                    sorted_keys = sorted(o.keys(), key=lambda k: (type(k).__name__, repr(k)))
                for k in sorted_keys:
                    encode(k)
                    encode(o[k])
                return
            elif isinstance(o, set):
                out.append(TAG_SET)
                write_varint(len(o))
                try:
                    sorted_elems = sorted(o)
                except TypeError:
                    sorted_elems = sorted(o, key=lambda x: (type(x).__name__, repr(x)))
                for item in sorted_elems:
                    encode(item)
                return

        if o is None:
            out.append(TAG_NULL)
        elif isinstance(o, bool):
            out.append(TAG_TRUE if o else TAG_FALSE)
        elif isinstance(o, int):
            out.append(TAG_INT)
            if o >= 0:
                zz = o * 2
            else:
                zz = -o * 2 - 1
            write_varint(zz)
        elif isinstance(o, float):
            out.append(TAG_FLOAT)
            out.extend(struct.pack('>d', o))
        elif isinstance(o, str):
            out.append(TAG_STR)
            b = o.encode('utf-8')
            write_varint(len(b))
            out.extend(b)
        elif isinstance(o, (bytes, bytearray)):
            out.append(TAG_BYTES)
            b = bytes(o)
            write_varint(len(b))
            out.extend(b)
        else:
            raise TypeError(f"unsupported type: {type(o).__name__}")

    encode(obj)
    return bytes(out)
