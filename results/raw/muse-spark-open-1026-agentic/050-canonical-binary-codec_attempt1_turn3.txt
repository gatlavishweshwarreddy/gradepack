import struct

TAG_NONE = 0x00
TAG_FALSE = 0x01
TAG_TRUE = 0x02
TAG_INT = 0x03
TAG_FLOAT = 0x04
TAG_STR = 0x05
TAG_BYTES = 0x06
TAG_LIST = 0x07
TAG_DICT = 0x08
TAG_SET = 0x09
TAG_TUPLE = 0x0A
TAG_REF = 0x0B


def _encode_sort_bytes(obj):
    """Deterministic encoding without memo for canonical sorting."""
    out = bytearray()

    def wv(n):
        while n >= 0x80:
            out.append((n & 0x7F) | 0x80)
            n >>= 7
        out.append(n)

    def enc(o):
        if o is None:
            out.append(TAG_NONE)
        elif o is True:
            out.append(TAG_TRUE)
        elif o is False:
            out.append(TAG_FALSE)
        elif isinstance(o, int) and not isinstance(o, bool):
            out.append(TAG_INT)
            zz = (o << 1) if o >= 0 else ((-o << 1) - 1)
            wv(zz)
        elif isinstance(o, float):
            out.append(TAG_FLOAT)
            out.extend(struct.pack('>d', o))
        elif isinstance(o, str):
            out.append(TAG_STR)
            b = o.encode('utf-8')
            wv(len(b))
            out.extend(b)
        elif isinstance(o, bytes):
            out.append(TAG_BYTES)
            wv(len(o))
            out.extend(o)
        elif isinstance(o, list):
            out.append(TAG_LIST)
            wv(len(o))
            for x in o:
                enc(x)
        elif isinstance(o, tuple):
            out.append(TAG_TUPLE)
            wv(len(o))
            for x in o:
                enc(x)
        elif isinstance(o, dict):
            out.append(TAG_DICT)
            wv(len(o))
            items = sorted(o.items(), key=lambda kv: repr(kv[0]))
            for k, v in items:
                enc(k)
                enc(v)
        elif isinstance(o, set):
            out.append(TAG_SET)
            wv(len(o))
            elems = sorted(o, key=repr)
            for e in elems:
                enc(e)
        else:
            # fallback
            out.append(TAG_STR)
            b = repr(o).encode('utf-8')
            wv(len(b))
            out.extend(b)

    enc(obj)
    return bytes(out)
