from dtype import coerce_element


class Matrix:
    def __init__(self, rows):
        self.rows = [[coerce_element(v) for v in row] for row in rows]
        self.nrows = len(rows)
        self.ncols = len(rows[0]) if rows else 0

    def __matmul__(self, other):
        if self.ncols != other.nrows:
            raise ValueError("incompatible shapes")
        result = []
        for i in range(self.nrows):
            row = []
            for j in range(other.ncols):
                s = sum(self.rows[i][k] * other.rows[k][j]
                        for k in range(self.ncols))
                row.append(coerce_element(s))
            result.append(row)
        return Matrix(result)

    def scale(self, factor):
        return Matrix([[coerce_element(v * factor) for v in row]
                       for row in self.rows])

    def __eq__(self, other):
        return self.rows == other.rows

    def __repr__(self):
        return f"Matrix({self.rows!r})"

def coerce_element(v):
    """
    Convert v to an appropriate numeric type.
    Preserves fractional values instead of truncating to int.
    """
    # Handle bool explicitly (bool is subclass of int)
    if isinstance(v, bool):
        return int(v)
    # Preserve int without loss (avoid float precision issues for large ints)
    if isinstance(v, int):
        return int(v)
    # Preserve float fractional part
    if isinstance(v, float):
        if v.is_integer():
            return int(v)
        return float(v)
    # For strings and other types, try int first then float
    try:
        iv = int(v)
        try:
            fv = float(v)
        except (TypeError, ValueError, OverflowError):
            return iv
        if fv == iv:
            return iv
        return fv
    except (TypeError, ValueError, OverflowError):
        pass
    try:
        fv = float(v)
        if fv.is_integer():
            return int(fv)
        return fv
    except (TypeError, ValueError, OverflowError):
        return v
