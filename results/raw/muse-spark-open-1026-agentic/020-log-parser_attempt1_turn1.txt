import datetime


def duration_seconds(entries):
    """
    Return elapsed seconds between first and last entry.
    Returns 0 if fewer than 2 entries.
    Handles logs that span midnight by advancing the date
    whenever the clock time goes backwards.
    """
    if len(entries) < 2:
        return 0
    base = datetime.date(2000, 1, 1)
    first = datetime.datetime.combine(base, entries[0].time)

    current_date = base
    prev_time = entries[0].time
    last = first

    for e in entries[1:]:
        if e.time < prev_time:
            current_date += datetime.timedelta(days=1)
        last = datetime.datetime.combine(current_date, e.time)
        prev_time = e.time

    delta = last - first
    return delta.total_seconds()
