import re

def _tokenize(expr):
    tokens = []
    i = 0
    n = len(expr)
    while i < n:
        c = expr[i]
        if c.isspace():
            i += 1
            continue
        if c.isdigit() or c == '.':
            start = i
            has_dot = (c == '.')
            i += 1
            while i < n and (expr[i].isdigit() or (expr[i] == '.' and not has_dot)):
                if expr[i] == '.':
                    has_dot = True
                i += 1
            tokens.append(expr[start:i])
            continue
        if c.isalpha() or c == '_':
            start = i
            i += 1
            while i < n and (expr[i].isalnum() or expr[i] == '_'):
                i += 1
            tokens.append(expr[start:i])
            continue
        if c in '+-*/()':
            tokens.append(c)
            i += 1
            continue
        raise ValueError(f"Unexpected character: {c!r}")
    return tokens

def _is_number(tok):
    try:
        float(tok)
        return True
    except ValueError:
        return False

class _Parser:
    def __init__(self, tokens, variables):
        self.tokens = tokens
        self.pos = 0
        self.vars = variables

    def peek(self):
        if self.pos < len(self.tokens):
            return self.tokens[self.pos]
        return None

    def consume(self, expected=None):
        tok = self.peek()
        if tok is None:
            raise ValueError("Unexpected end of expression")
        if expected is not None and tok != expected:
            raise ValueError(f"Expected {expected!r} but got {tok!r}")
        self.pos += 1
        return tok

    def parse_expression(self):
        # expression := term (('+'|'-') term)*
        val = self.parse_term()
        while True:
            op = self.peek()
            if op == '+' or op == '-':
                self.consume()
                right = self.parse_term()
                if op == '+':
                    val = val + right
                else:
                    val = val - right
            else:
                break
        return val

    def parse_term(self):
        # term := factor (('*'|'/') factor)*
        val = self.parse_factor()
        while True:
            op = self.peek()
            if op == '*' or op == '/':
                self.consume()
                right = self.parse_factor()
                if op == '*':
                    val = val * right
                else:
                    val = val / right
            else:
                break
        return val

    def parse_factor(self):
        tok = self.peek()
        if tok is None:
            raise ValueError("Unexpected end of expression")
        if tok == '(':
            self.consume('(')
            val = self.parse_expression()
            self.consume(')')
            return val
        if tok == '-':
            self.consume('-')
            return -self.parse_factor()
        if tok == '+':
            self.consume('+')
            return self.parse_factor()
        if _is_number(tok):
            self.consume()
            return float(tok)
        # variable
        if tok not in ('+', '-', '*', '/', '(', ')'):
            self.consume()
            if tok in self.vars:
                return float(self.vars[tok])
            raise NameError(f"Unknown variable: {tok}")
        raise ValueError(f"Unexpected token: {tok}")

def evaluate(expression, variables=None, **kwargs):
    """
    Evaluate an arithmetic expression with correct precedence.

    Supports +, -, *, /, parentheses, unary +/-,
    and variable substitution.

    Backward compatible: can be called with just the expression.
    Variables can be passed as a dict positional arg or as kwargs:
        evaluate("x + 2", {"x": 5})
        evaluate("x + 2", x=5)
        evaluate("x + y * 2", {"x": 1, "y": 3})
    """
    if variables is None:
        variables = {}
    elif not isinstance(variables, dict):
        # allow other mapping types
        variables = dict(variables)
    else:
        variables = dict(variables)  # copy

    if kwargs:
        variables.update(kwargs)

    if not isinstance(expression, str):
        raise TypeError("expression must be a string")

    tokens = _tokenize(expression)
    if not tokens:
        raise ValueError("Empty expression")
    parser = _Parser(tokens, variables)
    result = parser.parse_expression()
    if parser.peek() is not None:
        raise ValueError(f"Unexpected token: {parser.peek()!r}")
    return float(result)
