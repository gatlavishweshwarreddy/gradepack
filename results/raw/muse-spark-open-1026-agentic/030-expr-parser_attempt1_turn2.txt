import re

def tokenize(expr):
    tokens = []
    i = 0
    n = len(expr)
    while i < n:
        c = expr[i]
        if c.isspace():
            i += 1
            continue
        if c.isdigit() or c == '.':
            start = i
            dot_seen = False
            while i < n and (expr[i].isdigit() or expr[i] == '.'):
                if expr[i] == '.':
                    if dot_seen:
                        break
                    dot_seen = True
                i += 1
            tokens.append(('NUMBER', expr[start:i]))
            continue
        if c.isalpha() or c == '_':
            start = i
            while i < n and (expr[i].isalnum() or expr[i] == '_'):
                i += 1
            tokens.append(('IDENT', expr[start:i]))
            continue
        if c == '+':
            tokens.append(('PLUS', '+'))
            i += 1
            continue
        if c == '-':
            tokens.append(('MINUS', '-'))
            i += 1
            continue
        if c == '*':
            tokens.append(('MUL', '*'))
            i += 1
            continue
        if c == '/':
            tokens.append(('DIV', '/'))
            i += 1
            continue
        if c == '(':
            tokens.append(('LPAREN', '('))
            i += 1
            continue
        if c == ')':
            tokens.append(('RPAREN', ')'))
            i += 1
            continue
        raise SyntaxError(f"Invalid character {c!r}")
    return tokens

class Parser:
    def __init__(self, tokens, variables):
        self.tokens = tokens
        self.pos = 0
        self.vars = variables if variables is not None else {}

    def peek(self):
        if self.pos < len(self.tokens):
            return self.tokens[self.pos]
        return ('EOF', None)

    def consume(self, expected_type=None):
        tok = self.peek()
        if expected_type and tok[0] != expected_type:
            raise SyntaxError(f"Expected {expected_type} got {tok[0]}")
        self.pos += 1
        return tok

    def parse_expression(self):
        # expression handles + and - (lowest precedence)
        left = self.parse_term()
        while True:
            tok = self.peek()
            if tok[0] == 'PLUS':
                self.consume('PLUS')
                right = self.parse_term()
                left = left + right
            elif tok[0] == 'MINUS':
                self.consume('MINUS')
                right = self.parse_term()
                left = left - right
            else:
                break
        return left

    def parse_term(self):
        # term handles * and / (higher precedence)
        left = self.parse_factor()
        while True:
            tok = self.peek()
            if tok[0] == 'MUL':
                self.consume('MUL')
                right = self.parse_factor()
                left = left * right
            elif tok[0] == 'DIV':
                self.consume('DIV')
                right = self.parse_factor()
                left = left / right
            else:
                break
        return left

    def parse_factor(self):
        tok = self.peek()
        if tok[0] == 'NUMBER':
            self.consume('NUMBER')
            return float(tok[1])
        elif tok[0] == 'IDENT':
            self.consume('IDENT')
            name = tok[1]
            if name in self.vars:
                return float(self.vars[name])
            else:
                raise NameError(f"Undefined variable: {name}")
        elif tok[0] == 'LPAREN':
            self.consume('LPAREN')
            val = self.parse_expression()
            self.consume('RPAREN')
            return val
        elif tok[0] == 'MINUS':
            self.consume('MINUS')
            return -self.parse_factor()
        elif tok[0] == 'PLUS':
            self.consume('PLUS')
            return self.parse_factor()
        else:
            raise SyntaxError(f"Unexpected token {tok}")

def evaluate(expr, variables=None, *args, **kwargs):
    # backward compat: allow variables to be passed via different names or positions
    if variables is None:
        for key in ('variables', 'vars', 'env', 'context', 'values', 'variables_dict', 'var', 'bindings'):
            if key in kwargs:
                variables = kwargs[key]
                break
    if variables is None and args:
        if isinstance(args[0], dict):
            variables = args[0]
    # also if variables passed as kwarg directly like x=1 (unlikely but handle)
    # if still None, check if kwargs contains variable bindings (e.g., evaluate("x+1", x=1))
    if variables is None and kwargs:
        # treat remaining kwargs as variables if they look like bindings
        # but only if we didn't already find a dict
        # Heuristic: if kwargs not empty and no known key, use them
        variables = kwargs
    if variables is None:
        variables = {}
    if not isinstance(variables, dict):
        try:
            variables = dict(variables)
        except Exception:
            variables = {}

    tokens = tokenize(expr)
    if not tokens:
        raise ValueError("Empty expression")
    parser = Parser(tokens, variables)
    result = parser.parse_expression()
    if parser.pos != len(tokens):
        raise SyntaxError("Unexpected tokens at end of expression")
    return float(result)
