from datetime import datetime, timezone, timedelta
import zoneinfo


class EventScheduler:
    def __init__(self, clock):
        self._clock = clock
        self._events: list[tuple[str, datetime]] = []

    def add_event(self, name: str, utc_dt: datetime) -> None:
        if utc_dt.tzinfo is None:
            raise ValueError("utc_dt must be timezone-aware")
        self._events.append((name, utc_dt))

    def next_event_in(self, tz_name: str) -> tuple[str, timedelta] | None:
        tz = zoneinfo.ZoneInfo(tz_name)
        now = self._clock.now()
        # filter strictly upcoming events (as instants)
        upcoming = [
            (name, dt) for name, dt in self._events
            if dt > now
        ]
        if not upcoming:
            return None
        upcoming.sort(key=lambda x: x[1])
        name, dt = upcoming[0]
        # compute timedelta relative to current time in the given timezone
        now_in_tz = now.astimezone(tz)
        dt_in_tz = dt.astimezone(tz)
        delta = dt_in_tz - now_in_tz
        return name, delta
