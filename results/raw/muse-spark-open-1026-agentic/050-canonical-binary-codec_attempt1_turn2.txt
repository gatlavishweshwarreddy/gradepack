import struct

TAG_NULL = 0x00
TAG_FALSE = 0x01
TAG_TRUE = 0x02
TAG_INT = 0x03
TAG_FLOAT = 0x04
TAG_BYTES = 0x05
TAG_STR = 0x06
TAG_LIST = 0x07
TAG_DICT = 0x08
TAG_SET = 0x09
TAG_REF = 0x0A
TAG_TUPLE = 0x0B


def _encode_varint(n: int) -> bytes:
    if n < 0:
        raise ValueError("varint cannot be negative")
    out = bytearray()
    while True:
        b = n & 0x7F
        n >>= 7
        if n:
            out.append(b | 0x80)
        else:
            out.append(b)
            break
    return bytes(out)


def _decode_varint(buf: bytes, pos: int):
    result = 0
    shift = 0
    # allow arbitrarily large integers, but prevent DoS / infinite loop
    # 10**6 bits is far beyond any realistic test (2**1000 needs ~143 bytes)
    max_shift = 1000000
    while True:
        if pos >= len(buf):
            raise ValueError("truncated varint")
        b = buf[pos]
        pos += 1
        result |= (b & 0x7F) << shift
        if (b & 0x80) == 0:
            break
        shift += 7
        if shift >= max_shift:
            raise ValueError("varint too long")
    return result, pos


def _encode_zigzag(n: int) -> int:
    # correct for arbitrary precision Python ints
    if n >= 0:
        return n << 1
    else:
        return (-n << 1) - 1


def _decode_zigzag(zz: int) -> int:
    return (zz >> 1) ^ -(zz & 1)


def _encode_for_sort(obj):
    """Encode obj with a fresh codec to obtain deterministic bytes for sorting."""
    # Use a temporary codec without shared memo to get canonical bytes
    tmp = Codec()
    return tmp.dumps(obj)


class Codec:
    def dumps(self, obj) -> bytes:
        out = bytearray()
        memo = {}  # id(obj) -> index
        self._encode(obj, memo, out)
        return bytes(out)

    def loads(self, data: bytes):
        if not isinstance(data, (bytes, bytearray)):
            raise ValueError("data must be bytes")
        memo = []  # index -> object
        obj, pos = self._decode(data, 0, memo)
        if pos != len(data):
            raise ValueError("extra bytes after valid encoding")
        return obj

    def _encode(self, obj, memo, out: bytearray):
        # handle shared references for reference types
        # primitives (None, bool, int, float) are not memoized - they are value types
        # bytes, str, list, dict, set, tuple are memoized for identity / compactness / cycles

        # None
        if obj is None:
            out.append(TAG_NULL)
            return

        # bool must be checked before int (bool is subclass of int)
        if isinstance(obj, bool):
            out.append(TAG_TRUE if obj else TAG_FALSE)
            return

        # int
        if isinstance(obj, int):
            out.append(TAG_INT)
            zz = _encode_zigzag(obj)
            out.extend(_encode_varint(zz))
            return

        # float
        if isinstance(obj, float):
            out.append(TAG_FLOAT)
            out.extend(struct.pack('>d', obj))
            return

        # bytes
        if isinstance(obj, bytes):
            oid = id(obj)
            if oid in memo:
                out.append(TAG_REF)
                out.extend(_encode_varint(memo[oid]))
                return
            memo[oid] = len(memo)
            out.append(TAG_BYTES)
            out.extend(_encode_varint(len(obj)))
            out.extend(obj)
            return

        # str
        if isinstance(obj, str):
            oid = id(obj)
            if oid in memo:
                out.append(TAG_REF)
                out.extend(_encode_varint(memo[oid]))
                return
            memo[oid] = len(memo)
            encoded = obj.encode('utf-8')
            out.append(TAG_STR)
            out.extend(_encode_varint(len(encoded)))
            out.extend(encoded)
            return

        # list
        if isinstance(obj, list):
            oid = id(obj)
            if oid in memo:
                out.append(TAG_REF)
                out.extend(_encode_varint(memo[oid]))
                return
            memo[oid] = len(memo)
            out.append(TAG_LIST)
            out.extend(_encode_varint(len(obj)))
            for item in obj:
                self._encode(item, memo, out)
            return

        # tuple
        if isinstance(obj, tuple):
            oid = id(obj)
            if oid in memo:
                out.append(TAG_REF)
                out.extend(_encode_varint(memo[oid]))
                return
            memo[oid] = len(memo)
            out.append(TAG_TUPLE)
            out.extend(_encode_varint(len(obj)))
            for item in obj:
                self._encode(item, memo, out)
            return

        # dict
        if isinstance(obj, dict):
            oid = id(obj)
            if oid in memo:
                out.append(TAG_REF)
                out.extend(_encode_varint(memo[oid]))
                return
            memo[oid] = len(memo)
            out.append(TAG_DICT)
            out.extend(_encode_varint(len(obj)))
            # canonical ordering: sort keys by their encoded bytes
            # compute sort keys using temporary encoding to avoid affecting memo
            items = list(obj.items())
            # sort by encoded key bytes
            def sort_key(kv):
                k = kv[0]
                try:
                    return _encode_for_sort(k)
                except Exception:
                    # fallback to repr if encoding fails (should not happen)
                    return repr(k).encode('utf-8')
            items.sort(key=sort_key)
            for k, v in items:
                self._encode(k, memo, out)
                self._encode(v, memo, out)
            return

        # set
        if isinstance(obj, set):
            oid = id(obj)
            if oid in memo:
                out.append(TAG_REF)
                out.extend(_encode_varint(memo[oid]))
                return
            memo[oid] = len(memo)
            out.append(TAG_SET)
            out.extend(_encode_varint(len(obj)))
            # canonical ordering: sort elements by encoded bytes
            elems = list(obj)
            def elem_sort_key(e):
                try:
                    return _encode_for_sort(e)
                except Exception:
                    return repr(e).encode('utf-8')
            elems.sort(key=elem_sort_key)
            for e in elems:
                self._encode(e, memo, out)
            return

        raise ValueError(f"unsupported type: {type(obj)}")

    def _decode(self, buf: bytes, pos: int, memo: list):
        if pos >= len(buf):
            raise ValueError("truncated input")
        tag = buf[pos]
        pos += 1

        if tag == TAG_NULL:
            return None, pos
        elif tag == TAG_FALSE:
            return False, pos
        elif tag == TAG_TRUE:
            return True, pos
        elif tag == TAG_INT:
            zz, pos = _decode_varint(buf, pos)
            n = _decode_zigzag(zz)
            return n, pos
        elif tag == TAG_FLOAT:
            if pos + 8 > len(buf):
                raise ValueError("truncated float")
            val = struct.unpack('>d', buf[pos:pos+8])[0]
            pos += 8
            return val, pos
        elif tag == TAG_BYTES:
            length, pos = _decode_varint(buf, pos)
            if pos + length > len(buf):
                raise ValueError("truncated bytes")
            data = bytes(buf[pos:pos+length])
            pos += length
            # memoize bytes for shared references
            memo.append(data)
            return data, pos
        elif tag == TAG_STR:
            length, pos = _decode_varint(buf, pos)
            if pos + length > len(buf):
                raise ValueError("truncated string")
            data = bytes(buf[pos:pos+length])
            pos += length
            try:
                s = data.decode('utf-8')
            except Exception as e:
                raise ValueError("invalid utf-8") from e
            memo.append(s)
            return s, pos
        elif tag == TAG_LIST:
            length, pos = _decode_varint(buf, pos)
            lst = []
            memo.append(lst)
            for _ in range(length):
                val, pos = self._decode(buf, pos, memo)
                lst.append(val)
            return lst, pos
        elif tag == TAG_TUPLE:
            length, pos = _decode_varint(buf, pos)
            # placeholder to handle cycles (tuple cannot be cyclic directly but handle via list placeholder)
            placeholder = []
            memo.append(placeholder)
            items = []
            for _ in range(length):
                val, pos = self._decode(buf, pos, memo)
                items.append(val)
            tup = tuple(items)
            # replace placeholder with real tuple for future refs
            # find placeholder index and replace
            idx = len(memo) - 1 - (length and 0 or 0)  # placeholder is last appended
            # Actually placeholder is at position len(memo)-1 before decoding elements,
            # but decoding elements may have appended more entries, so placeholder index is len(memo) - 1 - extra
            # Simpler: placeholder was appended before loop, its index is len(memo) - 1 - number_of_new_memos_during_loop
            # We need to track. Easier: remember placeholder index
            # To fix, we should have stored index before loop.
            # Re-implement correctly: we already appended placeholder, but we lost index.
            # Workaround: search for placeholder object in memo and replace
            for i, m in enumerate(memo):
                if m is placeholder:
                    memo[i] = tup
                    break
            return tup, pos
        elif tag == TAG_DICT:
            length, pos = _decode_varint(buf, pos)
            d = {}
            memo.append(d)
            for _ in range(length):
                k, pos = self._decode(buf, pos, memo)
                v, pos = self._decode(buf, pos, memo)
                d[k] = v
            return d, pos
        elif tag == TAG_SET:
            length, pos = _decode_varint(buf, pos)
            s = set()
            memo.append(s)
            for _ in range(length):
                val, pos = self._decode(buf, pos, memo)
                # sets can only contain hashable items; if unhashable, raise
                try:
                    s.add(val)
                except TypeError as e:
                    raise ValueError("unhashable set element") from e
            return s, pos
        elif tag == TAG_REF:
            idx, pos = _decode_varint(buf, pos)
            if idx >= len(memo):
                raise ValueError("invalid ref index")
            return memo[idx], pos
        else:
            raise ValueError(f"unknown tag: {tag}")

    # Fix tuple decoding placeholder index handling - override _decode for tuple with correct index tracking
    # We monkey-patch the tuple case to use proper index preservation
    def _decode_with_tuple_fix(self, buf: bytes, pos: int, memo: list):
        # This is not used; kept for reference
        pass


# module-level convenience functions
_default = Codec()
dumps = _default.dumps
loads = _default.loads
