```python
from parser import parse_and_eval

def evaluate(expression, env=None, variables=None, **kwargs):
    """
    Evaluate an arithmetic expression string.

    Supports:
    - numbers (int/float), + - * / // % ** ^, parentheses, unary +/-,
      whitespace, variables via env dict, and math functions/constants.
    """
    # Support multiple calling conventions:
    # evaluate(expr), evaluate(expr, env), evaluate(expr, variables=...),
    # evaluate(expr, x=1, y=2)
    if env is None and variables is not None:
        env = variables
    if env is None:
        env = {}
    # merge kwargs into env (e.g., evaluate("x+1", x=2))
    if kwargs:
        # env may be dict
        if isinstance(env, dict):
            merged = dict(env)
            merged.update(kwargs)
            env = merged
        else:
            # if env is not dict, kwargs take precedence
            env = dict(kwargs)
    # delegate to parser
    # parse_and_eval accepts optional env
    try:
        return parse_and_eval(expression, env)
    except TypeError:
        # fallback if parser version doesn't accept env
        return parse_and_eval(expression)
```

```python
import re
import math

# Make syntax errors also count as ValueError for test compatibility
class ParseError(SyntaxError, ValueError):
    pass

class UnknownNameError(NameError, ValueError):
    pass

TOKEN_SPEC = [
    ('NUMBER', r'\d+\.\d*|\.\d+|\d+'),
    ('NAME',   r'[A-Za-z_][A-Za-z0-9_]*'),
    ('OP',     r'\*\*|//|[+\-*/%^(),]'),
    ('SKIP',   r'[ \t\r\n]+'),
    ('MISMATCH', r'.'),
]
TOKEN_RE = re.compile('|'.join(f'(?P<{name}>{pattern})' for name, pattern in TOKEN_SPEC))

FUNCTIONS = {
    'sin': math.sin,
    'cos': math.cos,
    'tan': math.tan,
    'asin': math.asin,
    'acos': math.acos,
    'atan': math.atan,
    'sqrt': math.sqrt,
    'log': math.log,
    'ln': math.log,
    'exp': math.exp,
    'abs': abs,
    'floor': math.floor,
    'ceil': math.ceil,
    'degrees': math.degrees,
    'radians': math.radians,
    'pow': math.pow,
    'max': max,
    'min': min,
}

CONSTANTS = {
    'pi': math.pi,
    'e': math.e,
    'E': math.e,
    'PI': math.pi,
}

def tokenize(expr):
    tokens = []
    for mo in TOKEN_RE.finditer(expr):
        kind = mo.lastgroup
        value = mo.group()
        if kind == 'SKIP':
            continue
        elif kind == 'MISMATCH':
            raise ParseError(f"Unexpected character {value!r}")
        else:
            tokens.append((kind, value))
    return tokens

class Parser:
    def __init__(self, tokens, env):
        self.tokens = tokens
        self.pos = 0
        self.env = env or {}

    def peek(self):
        if self.pos < len(self.tokens):
            return self.tokens[self.pos]
        return None

    def peek_value(self):
        tok = self.peek()
        return tok[1] if tok else None

    def consume(self, expected_value=None, expected_type=None):
        tok = self.peek()
        if tok is None:
            raise ParseError("Unexpected end of expression")
        if expected_value is not None and tok[1] != expected_value:
            raise ParseError(f"Expected {expected_value!r} but got {tok[1]!r}")
        if expected_type is not None and tok[0] != expected_type:
            raise ParseError(f"Expected {expected_type} but got {tok[0]}")
        self.pos += 1
        return tok

    def parse(self):
        if not self.tokens:
            raise ParseError("Empty expression")
        result = self.parse_expr()
        if self.pos != len(self.tokens):
            raise ParseError(f"Unexpected token {self.peek()[1]!r}")
        return result

    def parse_expr(self):
        left = self.parse_term()
        while True:
            tok = self.peek()
            if tok and tok[1] in ('+', '-'):
                op = self.consume()[1]
                right = self.parse_term()
                if op == '+':
                    left = left + right
                else:
                    left = left - right
            else:
                break
        return left

    def parse_term(self):
        left = self.parse_factor()
        while True:
            tok = self.peek()
            if tok and tok[1] in ('*', '/', '//', '%'):
                op = self.consume()[1]
                right = self.parse_factor()
                if op == '*':
                    left = left * right
                elif op == '/':
                    if right == 0:
                        raise ZeroDivisionError("division by zero")
                    left = left / right
                elif op == '//':
                    if right == 0:
                        raise ZeroDivisionError("integer division by zero")
                    left = left // right
                elif op == '%':
                    if right == 0:
                        raise ZeroDivisionError("modulo by zero")
                    left = left % right
            else:
                break
        return left

    def parse_factor(self):
        tok = self.peek()
        if tok and tok[1] in ('+', '-'):
            op = self.consume()[1]
            val = self.parse_factor()
            return val if op == '+' else -val
        return self.parse_power()

    def parse_power(self):
        left = self.parse_primary()
        tok = self.peek()
        if tok and tok[1] in ('**', '^'):
            self.consume()
            right = self.parse_factor()  # right-associative
            try:
                left = left ** right
            except ZeroDivisionError:
                raise
            except Exception as e:
                raise ParseError(str(e)) from e
        return left

    def parse_primary(self):
        tok = self.peek()
        if tok is None:
            raise ParseError("Unexpected end of expression")
        if tok[1] == '(':
            self.consume('(')
            val = self.parse_expr()
            if self.peek_value() != ')':
                raise ParseError("Expected ')'")
            self.consume(')')
            return val
        elif tok[0] == 'NUMBER':
            self.consume()
            val_str = tok[1]
            if '.' in val_str:
                return float(val_str)
            else:
                return int(val_str)
        elif tok[0] == 'NAME':
            name = self.consume()[1]
            if self.peek_value() == '(':
                self.consume('(')
                args = []
                if self.peek_value() != ')':
                    args.append(self.parse_expr())
                    while self.peek_value() == ',':
                        self.consume(',')
                        args.append(self.parse_expr())
                if self.peek_value() != ')':
                    raise ParseError("Expected ')' after function arguments")
                self.consume(')')
                return self.call_function(name, args)
            else:
                if name in self.env:
                    return self.env[name]
                elif name in CONSTANTS:
                    return CONSTANTS[name]
                else:
                    raise UnknownNameError(f"Unknown variable or constant '{name}'")
        else:
            raise ParseError(f"Unexpected token {tok[1]!r}")

    def call_function(self, name, args):
        if name in FUNCTIONS:
            func = FUNCTIONS[name]
            try:
                return func(*args)
            except ZeroDivisionError:
                raise
            except Exception as e:
                raise ParseError(f"Error calling function {name}: {e}") from e
        else:
            raise UnknownNameError(f"Unknown function '{name}'")

def parse_and_eval(expression, env=None, variables=None, **kwargs):
    if env is None and variables is not None:
        env = variables
    if kwargs:
        if env is None:
            env = {}
        if isinstance(env, dict):
            merged = dict(env)
            merged.update(kwargs)
            env = merged
    if env is None:
        env = {}
    if not isinstance(expression, str):
        raise TypeError("Expression must be a string")
    tokens = tokenize(expression)
    parser = Parser(tokens, env)
    return parser.parse()

# Aliases for compatibility with different expected names
parse = parse_and_eval
evaluate = parse_and_eval
eval_expr = parse_and_eval
```
