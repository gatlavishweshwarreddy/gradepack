import copy


class Schema:
    def __init__(self):
        self._definitions = {}

    def load(self, definition):
        """Store a schema definition dict."""
        self._definition = definition

    def define(self, name, sub_schema):
        """Register a named sub-schema for $ref resolution."""
        self._definitions[name] = sub_schema

    def expand(self):
        """
        Resolve $ref references.
        """
        expanded = copy.deepcopy(self._definition)
        self._resolve(expanded)
        return expanded

    def _resolve(self, node):
        if isinstance(node, dict):
            if "$ref" in node:
                ref_name = node["$ref"]
                if ref_name in self._definitions:
                    ref_copy = copy.deepcopy(self._definitions[ref_name])
                    del node["$ref"]
                    node.update(ref_copy)
            for v in list(node.values()):
                self._resolve(v)
        elif isinstance(node, list):
            for item in node:
                self._resolve(item)

from schema import Schema


def validate(data, schema_obj):
    """
    Validate data dict against schema_obj.
    Returns list of error strings (empty = valid).
    """
    expanded = schema_obj.expand()
    errors = []
    required = expanded.get("required", [])
    properties = expanded.get("properties", {})
    for field in required:
        if field not in data:
            errors.append(f"missing required field: {field!r}")
    for field, value in data.items():
        if field in properties:
            expected_type = properties[field].get("type")
            if expected_type == "string" and not isinstance(value, str):
                errors.append(f"{field!r}: expected string, got {type(value).__name__}")
            elif expected_type == "integer" and not isinstance(value, int):
                errors.append(f"{field!r}: expected integer, got {type(value).__name__}")
    return errors
