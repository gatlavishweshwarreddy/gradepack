from cache import ResultCache


def _make_hashable(obj):
    # Convert potentially unhashable/mutable structures to a hashable snapshot
    if isinstance(obj, list):
        return tuple(_make_hashable(x) for x in obj)
    if isinstance(obj, tuple):
        return tuple(_make_hashable(x) for x in obj)
    if isinstance(obj, dict):
        return tuple(sorted((_make_hashable(k), _make_hashable(v)) for k, v in obj.items()))
    if isinstance(obj, set):
        return frozenset(_make_hashable(x) for x in obj)
    try:
        hash(obj)
        return obj
    except TypeError:
        # Fallback for other unhashable types
        return repr(obj)


class Pipeline:
    def __init__(self, transforms):
        self._transforms = transforms
        self._cache = ResultCache()

    def run(self, data):
        result = data
        for t in self._transforms:
            # Use content-based key instead of id() so mutation is detected
            key = (t.name, _make_hashable(result))
            if self._cache.has(key):
                result = self._cache.get(key)
            else:
                result = t.apply(result)
                self._cache.set(key, result)
        return result

import copy


class ResultCache:
    def __init__(self):
        self._store = {}

    def get(self, key):
        val = self._store.get(key)
        try:
            return copy.deepcopy(val)
        except Exception:
            return val

    def set(self, key, value):
        try:
            self._store[key] = copy.deepcopy(value)
        except Exception:
            self._store[key] = value

    def has(self, key):
        return key in self._store
