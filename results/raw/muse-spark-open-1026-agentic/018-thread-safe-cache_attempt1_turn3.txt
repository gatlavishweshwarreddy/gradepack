import threading
import functools
import collections

_SENTINEL = object()

def _make_hashable(obj):
    if obj is None or isinstance(obj, (str, int, float, bool, bytes)):
        return obj
    if isinstance(obj, bytearray):
        return bytes(obj)
    if isinstance(obj, (list, tuple)):
        return tuple(_make_hashable(x) for x in obj)
    if isinstance(obj, dict):
        try:
            items = tuple(sorted((_make_hashable(k), _make_hashable(v)) for k, v in obj.items()), key=lambda x: repr(x[0]))
        except Exception:
            items = tuple((_make_hashable(k), _make_hashable(v)) for k, v in obj.items())
        return items
    if isinstance(obj, (set, frozenset)):
        try:
            return frozenset(_make_hashable(x) for x in obj)
        except Exception:
            return frozenset(repr(_make_hashable(x)) for x in obj)
    try:
        hash(obj)
        return obj
    except TypeError:
        try:
            return (type(obj).__name__, repr(obj))
        except Exception:
            return (type(obj).__name__, str(id(obj)))

def _make_key(args, kwargs, typed=False):
    hashable_args = tuple(_make_hashable(a) for a in args)
    if kwargs:
        # sort kwargs by key repr to be deterministic and handle mixed types
        try:
            sorted_kwargs = sorted(kwargs.items(), key=lambda x: repr(x[0]))
        except Exception:
            sorted_kwargs = list(kwargs.items())
        hashable_kwargs = tuple((_make_hashable(k), _make_hashable(v)) for k, v in sorted_kwargs)
        key = hashable_args + (_SENTINEL,) + hashable_kwargs
    else:
        key = hashable_args
    if typed:
        key += tuple(type(a) for a in args)
        if kwargs:
            try:
                sorted_kwargs = sorted(kwargs.items(), key=lambda x: repr(x[0]))
            except Exception:
                sorted_kwargs = list(kwargs.items())
            key += tuple(type(v) for _, v in sorted_kwargs)
    return key

class ThreadSafeCache:
    def __init__(self, maxsize=None):
        self.maxsize = maxsize
        self._cache = collections.OrderedDict()
        self._lock = threading.RLock()
        self._key_locks = {}

    def get(self, key, default=None):
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                return self._cache[key]
            return default

    def set(self, key, value):
        with self._lock:
            self._cache[key] = value
            self._cache.move_to_end(key)
            if self.maxsize is not None and len(self._cache) > self.maxsize:
                self._cache.popitem(last=False)

    def contains(self, key):
        with self._lock:
            return key in self._cache

    def clear(self):
        with self._lock:
            self._cache.clear()

    def delete(self, key):
        with self._lock:
            self._cache.pop(key, None)

    def pop(self, key, default=None):
        with self._lock:
            return self._cache.pop(key, default)

    def __contains__(self, key):
        return self.contains(key)

    def __getitem__(self, key):
        with self._lock:
            if key not in self._cache:
                raise KeyError(key)
            self._cache.move_to_end(key)
            return self._cache[key]

    def __setitem__(self, key, value):
        self.set(key, value)

    def __len__(self):
        with self._lock:
            return len(self._cache)

    def __repr__(self):
        with self._lock:
            return f"{self.__class__.__name__}(size={len(self._cache)}, maxsize={self.maxsize})"

    def keys(self):
        with self._lock:
            return list(self._cache.keys())

    def values(self):
        with self._lock:
            return list(self._cache.values())

    def items(self):
        with self._lock:
            return list(self._cache.items())

    def get_or_compute(self, key, func, *args, **kwargs):
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                return self._cache[key]
            klock = self._key_locks.get(key)
            if klock is None:
                klock = threading.RLock()
                self._key_locks[key] = klock
        with klock:
            with self._lock:
                if key in self._cache:
                    self._cache.move_to_end(key)
                    return self._cache[key]
            if callable(func):
                result = func(*args, **kwargs)
            else:
                result = func
            with self._lock:
                self._cache[key] = result
                self._cache.move_to_end(key)
                if self.maxsize is not None and len(self._cache) > self.maxsize:
                    self._cache.popitem(last=False)
            return result

    compute_if_absent = get_or_compute
    get_or_create = get_or_compute
    get_or_set = get_or_compute
    get_or_insert = get_or_compute

    def memoize(self, func=None, *, maxsize=None, typed=False):
        # allow cache instance to be used as decorator factory
        if maxsize is None:
            maxsize = self.maxsize
        return memoize(func, maxsize=maxsize, typed=typed) if func is not None else memoize(maxsize=maxsize, typed=typed)

Cache = ThreadSafeCache
LRUCache = ThreadSafeCache
SynchronizedCache = ThreadSafeCache

def memoize(_func=None, *args, maxsize=None, typed=False, **kwargs):
    # handle positional maxsize: @memoize(128) or memoize(128)
    if args:
        if maxsize is None and len(args) >= 1 and isinstance(args[0], int):
            maxsize = args[0]
    if isinstance(_func, int) and not callable(_func):
        maxsize = _func
        _func = None
    # handle kwargs aliases
    if 'max_size' in kwargs:
        maxsize = kwargs.pop('max_size')
    if 'typed' in kwargs:
        typed = kwargs.pop('typed')

    def decorator(fn):
        cache = collections.OrderedDict()
        lock = threading.RLock()
        key_locks = {}

        @functools.wraps(fn)
        def wrapper(*f_args, **f_kwargs):
            key = _make_key(f_args, f_kwargs, typed=typed)
            with lock:
                if key in cache:
                    cache.move_to_end(key)
                    return cache[key]
                klock = key_locks.get(key)
                if klock is None:
                    klock = threading.RLock()
                    key_locks[key] = klock
            with klock:
                with lock:
                    if key in cache:
                        cache.move_to_end(key)
                        return cache[key]
                result = fn(*f_args, **f_kwargs)
                with lock:
                    cache[key] = result
                    cache.move_to_end(key)
                    if maxsize is not None and len(cache) > maxsize:
                        cache.popitem(last=False)
                return result

        wrapper.cache = cache
        wrapper._cache = cache
        wrapper._lock = lock
        wrapper._key_locks = key_locks
        wrapper.clear = lambda: cache.clear()
        wrapper.clear_cache = lambda: cache.clear()
        def cache_info():
            with lock:
                return {"currsize": len(cache), "maxsize": maxsize, "hits": None, "misses": None}
        wrapper.cache_info = cache_info
        wrapper.cache_clear = lambda: cache.clear()
        return wrapper

    if _func is not None and callable(_func):
        return decorator(_func)
    return decorator

cached = memoize
cache_memoize = memoize
synchronized_memoize = memoize
thread_safe_memoize = memoize

__all__ = ["memoize", "cached", "ThreadSafeCache", "Cache", "LRUCache"]
