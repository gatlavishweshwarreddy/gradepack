import threading
import functools

def _make_key(args, kwargs):
    if not kwargs:
        if len(args) == 1:
            try:
                hash(args[0])
                return args[0]
            except TypeError:
                pass
        try:
            hash(args)
            return args
        except TypeError:
            return str(args)
    try:
        items = tuple(sorted(kwargs.items()))
        key = (args, items)
        hash(key)
        return key
    except TypeError:
        return str((args, kwargs))


class _MemoizedFunction:
    def __init__(self, fn):
        self.fn = fn
        self._cache = {}
        self._computing = {}
        self._lock = threading.Lock()
        functools.update_wrapper(self, fn)

    def __call__(self, *args, **kwargs):
        key = _make_key(args, kwargs)
        while True:
            with self._lock:
                if key in self._cache:
                    return self._cache[key]
                if key in self._computing:
                    event = self._computing[key]
                    is_owner = False
                else:
                    event = threading.Event()
                    self._computing[key] = event
                    is_owner = True
                    break
            event.wait()

        try:
            result = self.fn(*args, **kwargs)
        except BaseException:
            with self._lock:
                del self._computing[key]
                event.set()
            raise
        else:
            with self._lock:
                self._cache[key] = result
                del self._computing[key]
                event.set()
            return result

    def __get__(self, instance, owner):
        if instance is None:
            return self
        return functools.partial(self.__call__, instance)

    def cache_clear(self):
        with self._lock:
            self._cache.clear()

    def clear(self):
        self.cache_clear()

    @property
    def cache(self):
        return self._cache


def memoize(fn=None, **_kwargs):
    if fn is not None and callable(fn):
        return _MemoizedFunction(fn)
    def decorator(f):
        return _MemoizedFunction(f)
    return decorator
