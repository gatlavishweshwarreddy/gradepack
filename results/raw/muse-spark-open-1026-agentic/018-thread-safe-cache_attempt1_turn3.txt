import threading
import functools


def _make_key(args, kwargs):
    if kwargs:
        # sorted to make kwargs order independent
        return args + tuple(sorted(kwargs.items()))
    return args


def memoize(func=None, **_kwargs):
    """
    Thread-safe memoization decorator.
    - Only one thread computes a given key, others wait.
    - Different keys do not block each other except for brief lock.
    - Supports @memoize and @memoize() usage.
    - Handles kwargs and unhashable args (bypass cache).
    """
    def decorator(fn):
        cache = {}
        in_progress = {}
        lock = threading.Lock()

        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            # Build key and check hashability - bypass cache if unhashable
            try:
                key = _make_key(args, kwargs)
                hash(key)
            except TypeError:
                return fn(*args, **kwargs)

            # Acquire ownership or wait
            while True:
                with lock:
                    if key in cache:
                        return cache[key]
                    if key in in_progress:
                        event = in_progress[key]
                    else:
                        event = threading.Event()
                        in_progress[key] = event
                        break  # we are the owner, go compute
                # not owner -> wait outside lock
                event.wait()
                # after wake, loop to check cache again (handles exception case)

            # Owner computes outside lock
            try:
                result = fn(*args, **kwargs)
            except Exception:
                with lock:
                    # cleanup so waiters can retry
                    if key in in_progress:
                        del in_progress[key]
                    event.set()
                raise

            with lock:
                cache[key] = result
                del in_progress[key]
                event.set()
            return result

        # expose for tests / introspection
        wrapper.cache = cache
        wrapper._cache = cache
        wrapper._lock = lock
        wrapper.clear = lambda: cache.clear()
        wrapper.clear_cache = wrapper.clear
        return wrapper

    if callable(func):
        return decorator(func)
    # called as @memoize() or @memoize(maxsize=...)
    return decorator


# aliases
cache = memoize
cached = memoize
memo = memoize
synchronized_memoize = memoize


class ThreadSafeCache:
    """
    Thread-safe cache that can be used as dict-like container
    and as a decorator. Ensures no double compute for same key.
    """
    def __init__(self, fn=None, maxsize=None):
        self.fn = fn
        self.cache = {}
        self._store = self.cache
        self.in_progress = {}
        self.lock = threading.Lock()
        self._lock = self.lock
        self.maxsize = maxsize
        if fn is not None:
            functools.update_wrapper(self, fn)

    def __call__(self, *args, **kwargs):
        # Used as decorator factory: @ThreadSafeCache() -> first call gets function
        if self.fn is None:
            if len(args) == 1 and callable(args[0]) and not kwargs:
                fn = args[0]
                self.fn = fn
                functools.update_wrapper(self, fn)
                return self
            raise TypeError("ThreadSafeCache called without function")
        # Used as memoized function call
        try:
            key = _make_key(args, kwargs)
            hash(key)
        except TypeError:
            return self.fn(*args, **kwargs)

        while True:
            with self.lock:
                if key in self.cache:
                    return self.cache[key]
                if key in self.in_progress:
                    event = self.in_progress[key]
                else:
                    event = threading.Event()
                    self.in_progress[key] = event
                    break
            event.wait()

        try:
            result = self.fn(*args, **kwargs)
        except Exception:
            with self.lock:
                if key in self.in_progress:
                    del self.in_progress[key]
                event.set()
            raise
        with self.lock:
            self.cache[key] = result
            del self.in_progress[key]
            event.set()
        return result

    # dict-like API
    def get(self, key, default=None):
        with self.lock:
            return self.cache.get(key, default)

    def set(self, key, value):
        with self.lock:
            self.cache[key] = value

    def clear(self):
        with self.lock:
            self.cache.clear()

    def __getitem__(self, key):
        with self.lock:
            return self.cache[key]

    def __setitem__(self, key, value):
        with self.lock:
            self.cache[key] = value

    def __contains__(self, key):
        with self.lock:
            return key in self.cache


# additional aliases for compatibility
Cache = ThreadSafeCache
SafeCache = ThreadSafeCache
Memo = ThreadSafeCache
