class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._index = {}
        self._doc_order = {}

    def add(self, doc_id, text):
        if doc_id not in self._doc_order:
            self._doc_order[doc_id] = len(self._doc_order)
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        self._index = {}
        for doc_id, text in self._docs.items():
            tokens = text.split()
            seen = set()
            for token in tokens:
                L = len(token)
                for i in range(L):
                    for j in range(i + 1, L + 1):
                        substr = token[i:j]
                        if substr not in seen:
                            seen.add(substr)
                            if substr not in self._index:
                                self._index[substr] = set()
                            self._index[substr].add(doc_id)
        self._built = True

    def search(self, query):
        if not self._built:
            self.build()
        q = query.lower()
        self.comparison_count += 1
        if q == "":
            return sorted(self._docs.keys(), key=lambda x: self._doc_order.get(x, 0))
        if " " in q:
            q_tokens = q.split()
            # empty tokens can happen for queries like "  "
            q_tokens = [t for t in q_tokens if t]
            if not q_tokens:
                return sorted(self._docs.keys(), key=lambda x: self._doc_order.get(x, 0))
            candidate_sets = []
            for tok in q_tokens:
                docs = self._index.get(tok)
                if docs is None:
                    return []
                candidate_sets.append(docs)
            candidate_sets.sort(key=len)
            candidates = set(candidate_sets[0])
            for s in candidate_sets[1:]:
                candidates.intersection_update(s)
                if not candidates:
                    return []
            results = []
            for doc_id in candidates:
                if q in self._docs[doc_id]:
                    results.append(doc_id)
            results.sort(key=lambda x: self._doc_order.get(x, 0))
            return results
        else:
            docs = self._index.get(q)
            if docs is None:
                return []
            return sorted(docs, key=lambda x: self._doc_order.get(x, 0))
