def _encode_varint(n):
    out = bytearray()
    while n >= 0x80:
        out.append((n & 0x7F) | 0x80)
        n >>= 7
    out.append(n & 0x7F)
    return bytes(out)


def _decode_varint(buf, pos):
    result = 0
    shift = 0
    while True:
        if pos >= len(buf):
            raise ValueError("truncated varint")
        b = buf[pos]
        pos += 1
        result |= (b & 0x7F) << shift
        if (b & 0x80) == 0:
            break
        shift += 7
        if shift >= 64:
            raise ValueError("varint too long")
    return result, pos


def _encode_zigzag(n):
    if n >= 0:
        return n << 1
    else:
        return (-n << 1) - 1


def _decode_zigzag(n):
    return (n >> 1) ^ -(n & 1)


def _sort_key(obj):
    # Canonical total order for dict keys / set elements.
    # Order by type tag order, then by value. Handles heterogeneous types.
    if obj is None:
        return (0, 0)
    elif isinstance(obj, bool):
        return (1, int(obj))
    elif isinstance(obj, int):
        return (2, obj)
    elif isinstance(obj, float):
        # use IEEE-754 bit pattern for deterministic ordering (NaN, -0.0)
        return (3, struct.unpack('<Q', struct.pack('<d', obj))[0])
    elif isinstance(obj, str):
        return (4, obj)
    elif isinstance(obj, bytes):
        return (5, obj)
    elif isinstance(obj, list):
        return (6, len(obj))
    elif isinstance(obj, dict):
        return (7, len(obj))
    elif isinstance(obj, set):
        return (8, len(obj))
    else:
        return (9, repr(obj))
