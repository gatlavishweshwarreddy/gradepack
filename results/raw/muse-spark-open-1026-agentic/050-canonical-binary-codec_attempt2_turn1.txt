def _encode_varint(n):
    out = bytearray()
    while n >= 0x80:
        out.append((n & 0x7F) | 0x80)
        n >>= 7
    out.append(n & 0x7F)
    return bytes(out)


def _decode_varint(buf, pos):
    result = 0
    shift = 0
    start = pos
    while True:
        if pos >= len(buf):
            raise ValueError("truncated varint")
        b = buf[pos]
        pos += 1
        result |= (b & 0x7F) << shift
        if (b & 0x80) == 0:
            break
        shift += 7
        if shift >= 64:
            raise ValueError("varint too long")
    # canonical check: minimal encoding
    encoded = _encode_varint(result)
    if len(encoded) != (pos - start) or encoded != buf[start:pos]:
        raise ValueError("non-canonical varint")
    return result, pos


def _encode_zigzag(n):
    # canonical zigzag for arbitrary Python ints
    if n >= 0:
        return n << 1
    else:
        return (-n << 1) - 1


def _decode_zigzag(n):
    return (n >> 1) ^ -(n & 1)
