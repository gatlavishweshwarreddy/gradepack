import threading
import functools


def memoize(fn):
    """Cache results of fn. Thread-safe."""
    cache = {}
    lock = threading.RLock()

    @functools.wraps(fn)
    def wrapper(*args):
        with lock:
            if args in cache:
                return cache[args]
            result = fn(*args)
            cache[args] = result
            return result

    return wrapper
