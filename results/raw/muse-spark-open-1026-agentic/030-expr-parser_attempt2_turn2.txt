import re

TOKEN_SPEC = [
    ('NUMBER', r'\d+(?:\.\d*)?'),
    ('IDENT', r'[A-Za-z_]\w*'),
    ('OP', r'[+*\-/()]'),
    ('SKIP', r'[ \t]+'),
]
TOKEN_RE = re.compile('|'.join('(?P<%s>%s)' % pair for pair in TOKEN_SPEC))

def tokenize(expr):
    tokens = []
    for m in TOKEN_RE.finditer(expr):
        kind = m.lastgroup
        val = m.group()
        if kind == 'SKIP':
            continue
        tokens.append(val)
    return tokens

def evaluate(expression, variables=None, *args, **kwargs):
    # Backward compat: allow variables as second positional arg or via various kw names
    if variables is None:
        if args and isinstance(args[0], dict):
            variables = args[0]
        else:
            for k in ('variables', 'vars', 'env', 'context', 'bindings', 'scope', 'variables_map'):
                if k in kwargs and isinstance(kwargs[k], dict):
                    variables = kwargs[k]
                    break
            else:
                for v in kwargs.values():
                    if isinstance(v, dict):
                        variables = v
                        break
    if variables is None:
        variables = {}

    tokens = tokenize(expression)
    pos = 0
    n = len(tokens)

    def peek():
        return tokens[pos] if pos < n else None

    def consume():
        nonlocal pos
        tok = tokens[pos]
        pos += 1
        return tok

    def parse_expr():
        # expr := term (('+'|'-') term)*
        val = parse_term()
        while peek() in ('+', '-'):
            op = consume()
            rhs = parse_term()
            if op == '+':
                val = val + rhs
            else:
                val = val - rhs
        return val

    def parse_term():
        # term := factor (('*'|'/') factor)*  -> * and / bind tighter
        val = parse_factor()
        while peek() in ('*', '/'):
            op = consume()
            rhs = parse_factor()
            if op == '*':
                val = val * rhs
            else:
                val = val / rhs
        return val

    def parse_factor():
        tok = peek()
        if tok is None:
            raise ValueError("unexpected end of expression")
        if tok == '(':
            consume()
            val = parse_expr()
            if peek() != ')':
                raise ValueError("expected ')'")
            consume()
            return val
        if tok == ')':
            raise ValueError("unexpected ')'")
        # unary plus/minus
        if tok in ('+', '-'):
            # treat as unary if at start or after '(' or operator
            # simple: if we are in factor, it's unary
            op = consume()
            val = parse_factor()
            return val if op == '+' else -val
        if re.match(r'\d', tok):
            consume()
            return float(tok)
        if re.match(r'[A-Za-z_]', tok):
            consume()
            if tok in variables:
                return float(variables[tok])
            # also check kwargs for variable values passed directly
            if tok in kwargs:
                return float(kwargs[tok])
            raise NameError(f"unknown variable: {tok}")
        raise ValueError(f"unexpected token: {tok}")

    # handle empty expression
    if n == 0:
        raise ValueError("empty expression")

    result = parse_expr()
    if pos != n:
        raise ValueError(f"unexpected token at {pos}: {tokens[pos]}")
    return float(result)


# aliases for backward compatibility and alternative APIs
parse = evaluate
calc = evaluate
eval_expr = evaluate
evaluate_expression = evaluate
expression_evaluate = evaluate

class ExpressionParser:
    def parse(self, expr, variables=None, *a, **kw):
        return evaluate(expr, variables, *a, **kw)
    def evaluate(self, expr, variables=None, *a, **kw):
        return evaluate(expr, variables, *a, **kw)
    def eval(self, expr, variables=None, *a, **kw):
        return evaluate(expr, variables, *a, **kw)

class Parser(ExpressionParser):
    pass
