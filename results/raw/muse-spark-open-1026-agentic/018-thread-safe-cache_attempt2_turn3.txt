import threading
import time
from collections import OrderedDict

class ThreadSafeCache:
    def __init__(self, maxsize=128):
        self.maxsize = maxsize
        self._cache = OrderedDict()
        self._lock = threading.RLock()
        # per-key locks to prevent double compute
        self._key_locks = {}
        self._key_locks_lock = threading.Lock()

    def _get_key_lock(self, key):
        with self._key_locks_lock:
            if key not in self._key_locks:
                self._key_locks[key] = threading.Lock()
            return self._key_locks[key]

    def get(self, key):
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
                return self._cache[key]
            return None

    def set(self, key, value):
        with self._lock:
            if key in self._cache:
                self._cache.move_to_end(key)
            self._cache[key] = value
            if len(self._cache) > self.maxsize:
                self._cache.popitem(last=False)

    def memoize(self, func):
        def wrapper(*args, **kwargs):
            # Create a hashable key from args and kwargs
            key = (args, tuple(sorted(kwargs.items())))
            # Fast path: check cache under lock
            with self._lock:
                if key in self._cache:
                    self._cache.move_to_end(key)
                    return self._cache[key]

            # Acquire per-key lock to ensure only one thread computes
            key_lock = self._get_key_lock(key)
            with key_lock:
                # Double-checked locking: re-check cache after acquiring key lock
                with self._lock:
                    if key in self._cache:
                        self._cache.move_to_end(key)
                        return self._cache[key]

                # Compute outside global lock but inside key lock
                result = func(*args, **kwargs)

                # Store result
                with self._lock:
                    if key in self._cache:
                        self._cache.move_to_end(key)
                    self._cache[key] = result
                    if len(self._cache) > self.maxsize:
                        self._cache.popitem(last=False)
                    return result
            # Fallback (should not reach here normally)
            return result

        # Preserve metadata
        wrapper.__wrapped__ = func
        wrapper.__name__ = getattr(func, '__name__', 'wrapper')
        return wrapper

    def clear(self):
        with self._lock:
            self._cache.clear()

    def __contains__(self, key):
        with self._lock:
            return key in self._cache

    def __len__(self):
        with self._lock:
            return len(self._cache)
