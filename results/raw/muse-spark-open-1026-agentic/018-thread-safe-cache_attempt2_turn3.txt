import threading
import functools
from collections import namedtuple

CacheInfo = namedtuple('CacheInfo', ['hits', 'misses', 'maxsize', 'currsize'])

def _make_key(args, kwargs, typed=False):
    # Try to use functools._make_key for compatibility with Python's lru_cache
    try:
        return functools._make_key(args, kwargs, typed)
    except Exception:
        pass
    # Fallback: simple hashable key
    try:
        if kwargs:
            # sorted items to be deterministic
            items = tuple(sorted(kwargs.items()))
            key = (args, items)
        else:
            key = (args, ())
        # ensure hashable, otherwise fallback to repr
        hash(key)
        return key
    except TypeError:
        # unhashable args (e.g. list) -> use repr
        try:
            return (repr(args), repr(sorted(kwargs.items())) if kwargs else ())
        except Exception:
            return (id(args), id(kwargs))
