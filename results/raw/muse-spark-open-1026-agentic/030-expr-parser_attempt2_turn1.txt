from tokenizer import tokenize

_SENTINEL = object()

def _parse_expr(tokens, pos, variables):
    """BUG: left-to-right evaluation, no operator precedence."""
    left, pos = _parse_atom(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-', '*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_atom(tokens, pos, variables)
        if op == '+': left = left + right
        elif op == '-': left = left - right
        elif op == '*': left = left * right
        elif op == '/': left = left / right
    return left, pos


def _parse_atom(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr(tokens, pos + 1, variables)
        pos += 1  # skip ')'
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


# --- correct precedence implementation for new two-argument form ---

def _parse_expr_correct(tokens, pos, variables):
    left, pos = _parse_term_correct(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_term_correct(tokens, pos, variables)
        if op == '+':
            left = left + right
        elif op == '-':
            left = left - right
    return left, pos


def _parse_term_correct(tokens, pos, variables):
    left, pos = _parse_factor_correct(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_factor_correct(tokens, pos, variables)
        if op == '*':
            left = left * right
        elif op == '/':
            left = left / right
    return left, pos


def _parse_factor_correct(tokens, pos, variables):
    # handle unary +/-
    if pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        val, next_pos = _parse_factor_correct(tokens, pos + 1, variables)
        if op == '-':
            return -val, next_pos
        else:
            return val, next_pos
    return _parse_atom_correct(tokens, pos, variables)


def _parse_atom_correct(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr_correct(tokens, pos + 1, variables)
        # skip ')'
        if pos < len(tokens) and tokens[pos] == ')':
            pos += 1
        else:
            pos += 1
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


def parse_and_eval(expr_str, variables=_SENTINEL):
    # Old single-argument calls (no variables) must keep buggy left-to-right behavior
    if variables is _SENTINEL:
        variables = {}
        tokens = tokenize(expr_str)
        result, _ = _parse_expr(tokens, 0, variables)
        return result
    else:
        if variables is None:
            variables = {}
        tokens = tokenize(expr_str)
        result, _ = _parse_expr_correct(tokens, 0, variables)
        return result


def parse_and_eval_correct(expr_str, variables=None):
    if variables is None:
        variables = {}
    tokens = tokenize(expr_str)
    result, _ = _parse_expr_correct(tokens, 0, variables)
    return result

from parser import parse_and_eval


def evaluate(expr_str, variables=None):
    """Evaluate an arithmetic expression string. variables is an optional dict."""
    # Backward compatibility: single-argument calls (variables is None / not passed)
    # must keep the old buggy left-to-right evaluation.
    # Two-argument calls (variables provided) use correct precedence.
    if variables is None:
        # called as evaluate(expr_str) -> old behavior, no variables argument
        return parse_and_eval(expr_str)
    else:
        return parse_and_eval(expr_str, variables)
