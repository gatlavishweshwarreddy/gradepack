```python
from tokenizer import tokenize


def _parse_expr(tokens, pos, variables):
    """BUG: left-to-right evaluation, no operator precedence. Kept for backward compatibility."""
    left, pos = _parse_atom(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-', '*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_atom(tokens, pos, variables)
        if op == '+': left = left + right
        elif op == '-': left = left - right
        elif op == '*': left = left * right
        elif op == '/': left = left / right
    return left, pos


def _parse_atom(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr(tokens, pos + 1, variables)
        pos += 1  # skip ')'
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


# Correct precedence implementation for new two-argument form
def _parse_expr_correct(tokens, pos, variables):
    left, pos = _parse_term_correct(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_term_correct(tokens, pos, variables)
        if op == '+':
            left = left + right
        elif op == '-':
            left = left - right
    return left, pos


def _parse_term_correct(tokens, pos, variables):
    left, pos = _parse_atom_correct(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_atom_correct(tokens, pos, variables)
        if op == '*':
            left = left * right
        elif op == '/':
            left = left / right
    return left, pos


def _parse_atom_correct(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr_correct(tokens, pos + 1, variables)
        pos += 1  # skip ')'
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


def parse_and_eval(expr_str, variables=None):
    # variables is None means old single-argument form -> preserve buggy behavior
    # any other value (including {}) means new two-argument form -> correct precedence
    if variables is None:
        # backward compatible buggy path
        tokens = tokenize(expr_str)
        result, _ = _parse_expr(tokens, 0, {})
        return result
    else:
        tokens = tokenize(expr_str)
        result, _ = _parse_expr_correct(tokens, 0, variables)
        return result
```

tokenizer.py
```python
def tokenize(expr):
    """Split expression into list of tokens (numbers, operators, parens)."""
    import re
    tokens = []
    for m in re.finditer(r'\d+\.?\d*|[+\-*/()]|\b[a-zA-Z_]\w*\b', expr):
        tokens.append(m.group())
    return tokens
```

```python
from parser import parse_and_eval

_SENTINEL = object()

def evaluate(expr_str, variables=_SENTINEL):
    """Evaluate an arithmetic expression string. variables is an optional dict."""
    if variables is _SENTINEL:
        # old single-argument form - preserve backward compatibility (buggy precedence)
        return parse_and_eval(expr_str, None)
    else:
        # new two-argument form - correct precedence
        if variables is None:
            variables = {}
        return parse_and_eval(expr_str, variables)
```
