from tokenizer import tokenize


def _parse_expr(tokens, pos, variables):
    """Parse addition/subtraction (lowest precedence)."""
    left, pos = _parse_term(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_term(tokens, pos, variables)
        if op == '+':
            left = left + right
        elif op == '-':
            left = left - right
    return left, pos


def _parse_term(tokens, pos, variables):
    """Parse multiplication/division (higher precedence)."""
    left, pos = _parse_factor(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_factor(tokens, pos, variables)
        if op == '*':
            left = left * right
        elif op == '/':
            left = left / right
    return left, pos


def _parse_factor(tokens, pos, variables):
    """Parse unary plus/minus and delegate to atom."""
    if pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        val, pos = _parse_factor(tokens, pos, variables)
        if op == '-':
            val = -val
        return val, pos
    return _parse_atom(tokens, pos, variables)


def _parse_atom(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr(tokens, pos + 1, variables)
        # skip ')'
        if pos < len(tokens) and tokens[pos] == ')':
            pos += 1
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


def parse_and_eval(expr_str, variables=None):
    if variables is None:
        variables = {}
    tokens = tokenize(expr_str)
    if not tokens:
        raise ValueError("empty expression")
    result, _ = _parse_expr(tokens, 0, variables)
    return result
