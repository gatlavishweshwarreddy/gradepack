def encode(s):
    """Return run-length encoding as list of (char, count) tuples."""
    if not s:
        return []
    result = []
    prev_char = s[0]
    count = 1
    for ch in s[1:]:
        if ch == prev_char:
            count += 1
        else:
            result.append((prev_char, count))
            prev_char = ch
            count = 1
    result.append((prev_char, count))
    return result


def decode(pairs):
    """Reconstruct string from (char, count) pairs."""
    return "".join(ch * count for ch, count in pairs)
