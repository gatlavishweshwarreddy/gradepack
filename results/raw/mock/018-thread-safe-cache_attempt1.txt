import threading


def memoize(fn):
    """Cache results of fn. Thread-safe."""
    cache = {}
    lock = threading.Lock()

    def wrapper(*args):
        with lock:
            if args in cache:
                return cache[args]
        result = fn(*args)
        with lock:
            if args in cache:
                return cache[args]
            cache[args] = result
            return result

    return wrapper
