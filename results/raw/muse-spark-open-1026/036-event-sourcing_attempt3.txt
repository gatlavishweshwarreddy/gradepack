class EventStore:
    def __init__(self):
        self._events = []

    def append(self, event):
        self._events.append(event)

    def replay(self, from_seq=0):
        """
        Return events with sequence >= from_seq in sequence order.
        """
        return sorted([e for e in self._events if e.sequence >= from_seq], key=lambda e: e.sequence)

class Projector:
    def apply(self, events):
        """Fold events into state dict."""
        state = {}
        for e in sorted(events, key=lambda x: x.sequence):
            if e.type == "set":
                state[e.payload["key"]] = e.payload["value"]
            elif e.type == "delete":
                state.pop(e.payload["key"], None)
            elif e.type == "increment":
                k = e.payload["key"]
                state[k] = state.get(k, 0) + e.payload["by"]
        return state
