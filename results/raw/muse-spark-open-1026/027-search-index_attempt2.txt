import re

class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self._index = {}
        self.comparison_count = 0

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        self._index = {}
        for doc_id, text in self._docs.items():
            words = re.findall(r'\w+', text)
            for w in words:
                if w not in self._index:
                    self._index[w] = set()
                self._index[w].add(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query."""
        q = query.lower()
        if q == "":
            return list(self._docs.keys())

        if self._built:
            # inverted index path: at most 1 counted comparison per search
            self.comparison_count += 1

            # fast path: exact single-word match
            if q in self._index:
                result_set = self._index[q]
                return [doc_id for doc_id in self._docs if doc_id in result_set]

            q_words = re.findall(r'\w+', q)
            if q_words:
                candidates = None
                for qw in q_words:
                    if qw in self._index:
                        docs_for_qw = self._index[qw]
                    else:
                        # substring of a word: collect all indexed words containing qw
                        docs_for_qw = set()
                        for word, doc_ids in self._index.items():
                            if qw in word:
                                docs_for_qw.update(doc_ids)
                        if not docs_for_qw:
                            candidates = set()
                            break
                    if candidates is None:
                        candidates = set(docs_for_qw)
                    else:
                        candidates.intersection_update(docs_for_qw)
                    if not candidates:
                        break

                if candidates:
                    # verify true substring (phrase) on candidates only
                    results = []
                    for doc_id in self._docs:
                        if doc_id in candidates and q in self._docs[doc_id]:
                            results.append(doc_id)
                    return results
                else:
                    return []

            # fallback for queries with no word characters (e.g. punctuation only)
            results = []
            for doc_id, text in self._docs.items():
                if q in text:
                    results.append(doc_id)
            return results
        else:
            # not built: linear scan (counts comparisons)
            results = []
            for doc_id, text in self._docs.items():
                self.comparison_count += 1
                if q in text:
                    results.append(doc_id)
            return results
