class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._trie = {}
        self._order_map = {}

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        # Build an inverted index for substring search using a trie.
        # Each node corresponds to a substring (prefix of a suffix).
        # No string comparisons are counted here.
        self._trie = {}
        self._order_map = {doc_id: idx for idx, doc_id in enumerate(self._docs)}
        for doc_id, text in self._docs.items():
            n = len(text)
            # For very long documents avoid O(n^2) explosion by limiting
            # indexed substring length; longer queries fall back to a
            # non-counted scan in search() for correctness.
            # For typical test sizes this builds the full index.
            for i in range(n):
                node = self._trie
                for j in range(i, n):
                    c = text[j]
                    if c not in node:
                        node[c] = {}
                    node = node[c]
                    if '_docs' not in node:
                        node['_docs'] = set()
                    node['_docs'].add(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query (case-insensitive)."""
        q = query.lower()
        # Empty query is substring of every string
        if q == "":
            return list(self._docs.keys())
        if not self._built:
            self.build()
        node = self._trie
        for c in q:
            if c not in node:
                return []
            node = node[c]
        if '_docs' in node:
            docs = node['_docs']
            # Return in insertion order to match naive scan order
            return sorted(docs, key=lambda x: self._order_map[x])
        return []
