import unicodedata

def dedup(strings: list[str]) -> list[str]:
    """Return strings with duplicates removed, preserving first occurrence."""
    seen = set()
    result = []
    for s in strings:
        # Normalize to NFC for comparison so canonically equivalent
        # strings (e.g. precomposed vs base+combining) are treated as equal
        key = unicodedata.normalize('NFC', s)
        if key not in seen:
            seen.add(key)
            result.append(s)
    return result
