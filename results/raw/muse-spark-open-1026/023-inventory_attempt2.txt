from reservation import ReservationLog
import threading


class Inventory:
    def __init__(self):
        self._stock = {}
        self._log = ReservationLog()
        self._lock = threading.RLock()

    def add_stock(self, item, qty):
        with self._lock:
            self._stock[item] = self._stock.get(item, 0) + qty

    def available(self, item):
        with self._lock:
            return self._stock.get(item, 0) - self._log.total_reserved(item)

    def reserve(self, item, qty):
        # Atomically check available stock accounting for already committed reservations
        with self._lock:
            available = self._stock.get(item, 0) - self._log.total_reserved(item)
            if available >= qty:
                self._log.record(item, qty)
                return True
            return False

    def reservations(self):
        return self._log
