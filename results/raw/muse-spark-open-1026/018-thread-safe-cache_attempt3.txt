import threading
import functools


def memoize(fn):
    """Cache results of fn. Thread-safe for concurrent access."""
    cache = {}
    lock = threading.RLock()

    @functools.wraps(fn)
    def wrapper(*args, **kwargs):
        # Build a hashable key from args and kwargs
        if kwargs:
            key = (args, tuple(sorted(kwargs.items())))
        else:
            key = args
        with lock:
            if key in cache:
                return cache[key]
            result = fn(*args, **kwargs)
            cache[key] = result
            return result

    return wrapper
