import threading
import functools


def memoize(fn):
    """Cache results of fn. Thread-safe."""
    cache = {}
    lock = threading.Lock()

    @functools.wraps(fn)
    def wrapper(*args, **kwargs):
        if kwargs:
            # kwargs are sorted to make key deterministic and hashable
            key = (args, tuple(sorted(kwargs.items())))
        else:
            key = args
        with lock:
            if key in cache:
                return cache[key]
            result = fn(*args, **kwargs)
            cache[key] = result
            return result

    return wrapper
