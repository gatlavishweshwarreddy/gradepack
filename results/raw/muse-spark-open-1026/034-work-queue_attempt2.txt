import threading


class WorkQueue:
    def __init__(self):
        self._items = []
        self._lock = threading.Lock()
        self._done = False

    def put(self, item):
        with self._lock:
            self._items.append(item)

    def get(self):
        with self._lock:
            if self._items:
                return self._items.pop(0)
            return None

    def done(self):
        with self._lock:
            self._done = True

    def is_done(self):
        with self._lock:
            return self._done and len(self._items) == 0

import threading

_lock = threading.Lock()

class Worker:
    def __init__(self, queue, results):
        self._queue = queue
        self._results = results

    def run(self):
        while not self._queue.is_done():
            item = self._queue.get()
            if item is not None:
                with _lock:
                    self._results.append(item * 2)
