```python
def coerce(value, type_name):
    """
    Convert string value to type_name ('int', 'float', 'str').
    Raises ValueError for non-numeric values when type is int or float.
    """
    if type_name == "int":
        try:
            # Try direct int conversion first to preserve precision for large ints
            return int(value)
        except (ValueError, TypeError):
            pass
        # Fallback: convert via float first then truncate to handle '1.0', '42.0', etc.
        try:
            return int(float(value))
        except (ValueError, TypeError, OverflowError) as e:
            raise ValueError(str(e)) from e
    elif type_name == "float":
        return float(value)
    else:
        return str(value)
```

```python
from schema import validate_schema
from coerce import coerce


def import_csv(rows, schema):
    """
    Convert list-of-dicts rows using schema (col -> type name).
    Returns list of dicts with values coerced to the schema types.
    Raises ValueError for invalid values.
    """
    validate_schema(schema)
    result = []
    for row in rows:
        converted = {}
        for col, type_name in schema.items():
            raw = row.get(col, "")
            converted[col] = coerce(raw, type_name)
        result.append(converted)
    return result
```

schema.py
```python
VALID_TYPES = {"int", "float", "str"}


def validate_schema(schema):
    """Raise ValueError if any type name is not valid."""
    for col, type_name in schema.items():
        if type_name not in VALID_TYPES:
            raise ValueError(f"Unknown type {type_name!r} for column {col!r}")
```
