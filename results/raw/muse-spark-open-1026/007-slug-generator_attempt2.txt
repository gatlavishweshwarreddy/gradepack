import re


def slugify(text):
    """Convert text to a URL slug."""
    # Lowercase first
    text = text.lower()
    # Replace non-alphanumeric characters with hyphens
    slug = re.sub(r"[^a-z0-9]+", "-", text)
    # Collapse consecutive hyphens (covered by + above, but ensure explicitly)
    slug = re.sub(r"-+", "-", slug)
    # Remove leading/trailing hyphens
    slug = slug.strip("-")
    return slug
