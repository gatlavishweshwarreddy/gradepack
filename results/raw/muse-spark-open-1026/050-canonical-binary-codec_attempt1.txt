import struct

TAG_NULL = 0x00
TAG_TRUE = 0x01
TAG_FALSE = 0x02
TAG_INT = 0x03
TAG_FLOAT = 0x04
TAG_STR = 0x05
TAG_BYTES = 0x06
TAG_LIST = 0x07
TAG_DICT = 0x08
TAG_SET = 0x09
TAG_REF = 0x0A


def _encode_varint(n):
    out = bytearray()
    while n >= 0x80:
        out.append((n & 0x7F) | 0x80)
        n >>= 7
    out.append(n & 0x7F)
    return bytes(out)


def _decode_varint(buf, pos):
    result = 0
    shift = 0
    while True:
        if pos >= len(buf):
            raise ValueError("truncated varint")
        b = buf[pos]
        pos += 1
        result |= (b & 0x7F) << shift
        if (b & 0x80) == 0:
            break
        shift += 7
        if shift >= 64:
            raise ValueError("varint too long")
    return result, pos


def _encode_zigzag(n):
    # canonical ZigZag for arbitrary Python ints: (n<<1) ^ -(n<0)
    # equivalent to n*2 if n>=0 else -n*2-1
    return (n << 1) ^ (-1 if n < 0 else 0)


def _decode_zigzag(n):
    return (n >> 1) ^ -(n & 1)


# Helper for canonical sorting: encode value to bytes deterministically
# without memo, with sorted dict/set contents. Used as sort key.
def _encode_key(obj):
    buf = bytearray()
    _encode_key_inner(obj, buf)
    return bytes(buf)


def _encode_key_inner(obj, buf):
    if obj is None:
        buf.append(TAG_NULL)
    elif isinstance(obj, bool):
        buf.append(TAG_TRUE if obj else TAG_FALSE)
    elif isinstance(obj, int):
        buf.append(TAG_INT)
        zz = _encode_zigzag(obj)
        buf.extend(_encode_varint(zz))
    elif isinstance(obj, float):
        buf.append(TAG_FLOAT)
        buf.extend(struct.pack('<d', obj))
    elif isinstance(obj, str):
        b = obj.encode('utf-8')
        buf.append(TAG_STR)
        buf.extend(_encode_varint(len(b)))
        buf.extend(b)
    elif isinstance(obj, bytes):
        buf.append(TAG_BYTES)
        buf.extend(_encode_varint(len(obj)))
        buf.extend(obj)
    elif isinstance(obj, list):
        buf.append(TAG_LIST)
        buf.extend(_encode_varint(len(obj)))
        for item in obj:
            _encode_key_inner(item, buf)
    elif isinstance(obj, dict):
        buf.append(TAG_DICT)
        buf.extend(_encode_varint(len(obj)))
        # sort keys by their canonical encoding
        try:
            sorted_keys = sorted(obj.keys(), key=_encode_key)
        except Exception:
            # fallback for uncomparable types
            sorted_keys = sorted(obj.keys(), key=lambda x: (str(type(x)), str(x)))
        for k in sorted_keys:
            _encode_key_inner(k, buf)
            _encode_key_inner(obj[k], buf)
    elif isinstance(obj, set):
        buf.append(TAG_SET)
        buf.extend(_encode_varint(len(obj)))
        try:
            sorted_elems = sorted(obj, key=_encode_key)
        except Exception:
            sorted_elems = sorted(obj, key=lambda x: (str(type(x)), str(x)))
        for e in sorted_elems:
            _encode_key_inner(e, buf)
    else:
        raise TypeError(f"unsupported type: {type(obj)}")
