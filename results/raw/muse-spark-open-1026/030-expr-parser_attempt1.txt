```python
from tokenizer import tokenize

_SENTINEL = object()

def _parse_expr_buggy(tokens, pos, variables):
    """BUG: left-to-right evaluation, no operator precedence."""
    left, pos = _parse_atom_buggy(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-', '*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_atom_buggy(tokens, pos, variables)
        if op == '+': left = left + right
        elif op == '-': left = left - right
        elif op == '*': left = left * right
        elif op == '/': left = left / right
    return left, pos


def _parse_atom_buggy(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr_buggy(tokens, pos + 1, variables)
        pos += 1  # skip ')'
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


def _parse_expr_correct(tokens, pos, variables):
    left, pos = _parse_term_correct(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('+', '-'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_term_correct(tokens, pos, variables)
        if op == '+':
            left = left + right
        elif op == '-':
            left = left - right
    return left, pos


def _parse_term_correct(tokens, pos, variables):
    left, pos = _parse_atom_correct(tokens, pos, variables)
    while pos < len(tokens) and tokens[pos] in ('*', '/'):
        op = tokens[pos]
        pos += 1
        right, pos = _parse_atom_correct(tokens, pos, variables)
        if op == '*':
            left = left * right
        elif op == '/':
            left = left / right
    return left, pos


def _parse_atom_correct(tokens, pos, variables):
    tok = tokens[pos]
    if tok == '(':
        val, pos = _parse_expr_correct(tokens, pos + 1, variables)
        pos += 1  # skip ')'
        return val, pos
    elif tok in variables:
        return float(variables[tok]), pos + 1
    else:
        return float(tok), pos + 1


# Keep original names for backward compatibility (buggy)
def _parse_expr(tokens, pos, variables):
    return _parse_expr_buggy(tokens, pos, variables)


def _parse_atom(tokens, pos, variables):
    return _parse_atom_buggy(tokens, pos, variables)


def parse_and_eval(expr_str, variables=_SENTINEL):
    if variables is _SENTINEL:
        # old single-argument call: preserve buggy left-to-right behavior
        variables = {}
        tokens = tokenize(expr_str)
        result, _ = _parse_expr_buggy(tokens, 0, variables)
        return result
    else:
        if variables is None:
            variables = {}
        tokens = tokenize(expr_str)
        result, _ = _parse_expr_correct(tokens, 0, variables)
        return result
```

```python
from parser import parse_and_eval

_SENTINEL = object()

def evaluate(expr_str, variables=_SENTINEL):
    """Evaluate an arithmetic expression string. variables is an optional dict."""
    if variables is _SENTINEL:
        # called with one argument: backward compatibility, keep old buggy behavior
        return parse_and_eval(expr_str)
    else:
        if variables is None:
            variables = {}
        return parse_and_eval(expr_str, variables)
```
