```python
import threading

try:
    from reservation import ReservationLog
except ImportError:
    from .reservation import ReservationLog


class Inventory:
    def __init__(self):
        self._stock = {}
        self._log = ReservationLog()
        self._lock = threading.Lock()

    def add_stock(self, item, qty):
        with self._lock:
            self._stock[item] = self._stock.get(item, 0) + qty

    def available(self, item):
        with self._lock:
            return self._stock.get(item, 0) - self._log.total_reserved(item)

    def reserve(self, item, qty):
        with self._lock:
            # atomically check and record; available must account for already-committed reservations
            available = self._stock.get(item, 0) - self._log.total_reserved(item)
            if available >= qty:
                self._log.record(item, qty)
                return True
            return False

    def reservations(self):
        return self._log
```

```python
import threading

class ReservationLog:
    def __init__(self):
        self._log = []
        self._lock = threading.Lock()

    def record(self, item, qty):
        with self._lock:
            self._log.append((item, qty))

    def total_reserved(self, item):
        with self._lock:
            return sum(q for i, q in self._log if i == item)

    def all_reservations(self):
        with self._lock:
            return list(self._log)
```
