from decimal import Decimal, InvalidOperation


class Ledger:
    def __init__(self, initial_balance="0"):
        try:
            self._balance = Decimal(str(initial_balance))
        except (InvalidOperation, ValueError, TypeError) as exc:
            raise ValueError(f"Invalid amount: {initial_balance!r}") from exc

    def add(self, amount_str: str) -> None:
        try:
            self._balance += Decimal(amount_str)
        except (InvalidOperation, ValueError, TypeError, AttributeError) as exc:
            raise ValueError(f"Invalid amount: {amount_str!r}") from exc

    def balance(self) -> str:
        return f"{self._balance:.2f}"
