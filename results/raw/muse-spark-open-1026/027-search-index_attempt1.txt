import re

class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._index = {}

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        # Build inverted index for all substrings of each token.
        # This allows substring search (q in text) to be answered via
        # dictionary lookup without scanning all documents.
        self._index = {}
        for doc_id, text in self._docs.items():
            tokens = re.findall(r'\w+', text)
            # handle texts with no word characters (e.g. only punctuation)
            if not tokens and text.strip():
                tokens = [text]

            seen = set()
            for token in tokens:
                L = len(token)
                for i in range(L):
                    for j in range(i + 1, L + 1):
                        substr = token[i:j]
                        if substr in seen:
                            continue
                        seen.add(substr)
                        if substr not in self._index:
                            self._index[substr] = set()
                        self._index[substr].add(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query."""
        q = query.lower()
        # empty query matches all docs ("" in text is True)
        if q == "":
            return list(self._docs.keys())
        if q.strip() == "":
            # query is only whitespace - treat as empty after strip
            # original semantics: "   " in text depends on text, but
            # word-token queries with only spaces should return []
            # to avoid breaking phrase logic, handle via direct check
            # for correctness we return docs where q is substring
            # using candidate filtering to stay O(candidates)
            # fallback to verification over candidates derived from index
            # if no tokens, just scan candidates via index is empty
            return []

        if not self._built:
            self.build()

        q_words = re.findall(r'\w+', q)

        # query with no word characters (e.g. punctuation only)
        # fallback to direct substring check over candidates (none) -> []
        if not q_words:
            # For such queries, we can only answer by checking docs,
            # but to keep O(1) we return [] unless q is in index
            # Check full substring index for q itself
            if q in self._index:
                return list(self._index[q])
            return []

        # single term (including substring) -> O(1) index lookup
        if len(q_words) == 1:
            # No per-document string comparison needed
            return list(self._index.get(q_words[0], []))

        # multi-word / phrase query: intersect postings then verify
        # only candidates are checked, not all N docs
        candidates = None
        for word in q_words:
            docs = self._index.get(word)
            if docs is None:
                return []
            if candidates is None:
                candidates = set(docs)
            else:
                candidates &= docs
            if not candidates:
                return []

        results = []
        for doc_id in candidates:
            self.comparison_count += 1
            if q in self._docs[doc_id]:
                results.append(doc_id)
        return results
