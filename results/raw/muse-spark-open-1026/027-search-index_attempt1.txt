import re

class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._index = {}
        self._ngram_index = {}

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        self._index = {}
        self._ngram_index = {}
        word_pat = re.compile(r'\w+')
        for doc_id, text in self._docs.items():
            # word inverted index
            for term in word_pat.findall(text):
                if term not in self._index:
                    self._index[term] = set()
                self._index[term].add(doc_id)
            # n-gram index for 1..3 characters (including spaces/punctuation)
            L = len(text)
            for i in range(L):
                # 1-gram
                ng1 = text[i:i+1]
                if ng1:
                    self._ngram_index.setdefault(ng1, set()).add(doc_id)
                # 2-gram
                if i + 2 <= L:
                    ng2 = text[i:i+2]
                    self._ngram_index.setdefault(ng2, set()).add(doc_id)
                # 3-gram
                if i + 3 <= L:
                    ng3 = text[i:i+3]
                    self._ngram_index.setdefault(ng3, set()).add(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query."""
        if not self._built:
            self.build()
        q = query.lower()
        if q == "":
            # "" is substring of every string -> match all docs (original behaviour)
            return list(self._docs.keys())

        # exact word match via inverted index - O(1), no doc comparisons
        if q in self._index:
            result_set = self._index[q]
            return [doc_id for doc_id in self._docs if doc_id in result_set]

        # single-word substring inside a word via vocabulary scan
        # avoids any doc text comparison; if a term contains q, its docs contain q
        if ' ' not in q:
            matched = set()
            for term, docs in self._index.items():
                if q in term:
                    matched.update(docs)
            if matched:
                return [doc_id for doc_id in self._docs if doc_id in matched]

        # fallback for phrases / substrings spanning words / punctuation
        # use n-gram index to get candidates, then verify only candidates
        if len(q) == 1:
            ngrams = [q]
        elif len(q) == 2:
            ngrams = [q]
        else:
            ngrams = [q[i:i+3] for i in range(len(q) - 2)]

        candidates = None
        for ng in ngrams:
            posting = self._ngram_index.get(ng)
            if posting is None:
                return []
            if candidates is None:
                candidates = set(posting)
            else:
                candidates.intersection_update(posting)
                if not candidates:
                    return []

        if candidates is None:
            return []

        results = []
        for doc_id in self._docs:
            if doc_id not in candidates:
                continue
            text = self._docs[doc_id]
            self.comparison_count += 1
            if q in text:
                results.append(doc_id)
        return results
