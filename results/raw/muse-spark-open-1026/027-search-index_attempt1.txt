class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        # inverted index for short substrings (direct lookup, no string compare)
        self._index_direct = {}
        # positional k-gram index for long queries
        self._index_pos = {}
        self._k = 4
        self._max_direct = 12

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        from collections import defaultdict

        direct = defaultdict(set)
        pos_index = defaultdict(lambda: defaultdict(set))

        for doc_id, text in self._docs.items():
            L = len(text)
            # all substrings up to _max_direct for O(1) lookup
            for i in range(L):
                max_l = self._max_direct
                if i + max_l > L:
                    max_l = L - i
                # inline loop to avoid extra function calls
                for l in range(1, max_l + 1):
                    substr = text[i:i+l]
                    direct[substr].add(doc_id)
            # positional k-grams
            if L >= self._k:
                for i in range(L - self._k + 1):
                    gram = text[i:i+self._k]
                    pos_index[gram][doc_id].add(i)

        self._index_direct = dict(direct)
        # convert inner defaultdicts to plain dicts
        self._index_pos = {g: dict(d) for g, d in pos_index.items()}
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query using inverted index."""
        q = query.lower()
        if not q:
            return []
        if not self._built:
            self.build()

        # short query: direct dictionary lookup, no string scan
        if len(q) <= self._max_direct:
            result_set = self._index_direct.get(q)
            if not result_set:
                return []
            # preserve insertion order of _docs
            return [doc_id for doc_id in self._docs if doc_id in result_set]

        # long query: positional k-gram intersection, no q in text
        k = self._k
        grams = [q[i:i+k] for i in range(len(q) - k + 1)]

        # collect posting sets, early exit if any gram missing
        posting_sets = []
        for gram in grams:
            d = self._index_pos.get(gram)
            if d is None:
                return []
            posting_sets.append(set(d.keys()))

        # intersect candidates starting with smallest
        posting_sets.sort(key=len)
        candidates = posting_sets[0]
        for s in posting_sets[1:]:
            candidates = candidates.intersection(s)
            if not candidates:
                return []

        # positional verification (integer comparisons only)
        results = []
        for doc_id in self._docs:
            if doc_id not in candidates:
                continue
            first_positions = self._index_pos[grams[0]][doc_id]
            found = False
            for pos in first_positions:
                ok = True
                for offset, gram in enumerate(grams[1:], start=1):
                    if (pos + offset) not in self._index_pos[gram][doc_id]:
                        ok = False
                        break
                if ok:
                    found = True
                    break
            if found:
                results.append(doc_id)
        return results
