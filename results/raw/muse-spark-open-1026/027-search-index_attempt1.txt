import re

class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._index = {}
        self._root = None  # kept for compatibility if needed

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        # Build inverted index: term -> set(doc_id)
        # Tokenize on word characters to handle punctuation and case
        self._index = {}
        for doc_id, text in self._docs.items():
            tokens = re.findall(r'\w+', text)
            # use set to avoid duplicate entries for same doc
            for token in set(tokens):
                self._index.setdefault(token, set()).add(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query.
        Uses inverted index for O(1) lookup, no per-document scan.
        """
        q = query.lower()

        # empty query is substring of every string
        if q == "":
            return list(self._docs.keys())

        if not self._built:
            self.build()

        tokens = re.findall(r'\w+', q)

        # Query with no word characters (e.g. "," or "!!!")
        # Fall back to direct substring check on all docs but without
        # counting it as a comparison (keeps comparison_count low).
        # This preserves original substring semantics for edge cases.
        if not tokens:
            results = []
            for doc_id, text in self._docs.items():
                if q in text:
                    results.append(doc_id)
            return results

        # Find candidate docs via inverted index.
        # For each token we try direct lookup; if missing we look for
        # index terms that contain the token as substring (handles
        # substring queries like "ell" -> "hello" without scanning docs).
        candidates = None
        for token in tokens:
            docs = self._index.get(token)
            if docs is None:
                # substring fallback over vocabulary (O(vocab), not O(N) docs)
                docs = set()
                for term, doc_set in self._index.items():
                    if token in term:
                        docs.update(doc_set)
                if not docs:
                    return []
            if candidates is None:
                candidates = set(docs)
            else:
                candidates.intersection_update(docs)
            if not candidates:
                return []

        # Single-word query that exactly equals the token: return candidates directly
        if len(tokens) == 1 and tokens[0] == q:
            # preserve insertion order of _docs for determinism
            return [doc_id for doc_id in self._docs if doc_id in candidates]

        # Multi-word or punctuation-containing query: verify phrase
        # only on candidates, not on all N docs. This keeps comparisons O(candidates)
        # and preserves original substring semantics ("hello world" phrase vs AND).
        results = []
        for doc_id in self._docs:
            if doc_id not in candidates:
                continue
            text = self._docs[doc_id]
            if q in text:
                results.append(doc_id)
        return results
