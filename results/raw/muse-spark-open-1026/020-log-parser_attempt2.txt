import datetime


def duration_seconds(entries):
    """
    Return elapsed seconds between first and last entry.
    Returns 0 if fewer than 2 entries.
    Handles logs that span midnight.
    """
    if len(entries) < 2:
        return 0
    base = datetime.date(2000, 1, 1)
    total = 0.0
    for i in range(1, len(entries)):
        prev = datetime.datetime.combine(base, entries[i - 1].time)
        curr = datetime.datetime.combine(base, entries[i].time)
        delta = (curr - prev).total_seconds()
        if delta < 0:
            delta += 24 * 3600
        total += delta
    return total
