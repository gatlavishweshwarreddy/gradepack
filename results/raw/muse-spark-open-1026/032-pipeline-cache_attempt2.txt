from cache import ResultCache


def _to_hashable(value):
    # Recursively convert unhashable containers to hashable equivalents
    # so the cache key captures the full contents of the input.
    if isinstance(value, list):
        return tuple(_to_hashable(v) for v in value)
    elif isinstance(value, tuple):
        return tuple(_to_hashable(v) for v in value)
    elif isinstance(value, dict):
        return tuple(sorted((_to_hashable(k), _to_hashable(v)) for k, v in value.items()))
    elif isinstance(value, set):
        return frozenset(_to_hashable(v) for v in value)
    else:
        try:
            hash(value)
            return value
        except TypeError:
            # Fallback for unhashable custom objects
            return repr(value)


class Pipeline:
    def __init__(self, transforms):
        self._transforms = transforms
        self._cache = ResultCache()

    def run(self, data):
        result = data
        for t in self._transforms:
            # Cache key captures transform identity and full contents of input
            key = (t.name, _to_hashable(result))
            if self._cache.has(key):
                result = self._cache.get(key)
            else:
                result = t.apply(result)
                self._cache.set(key, result)
        return result
