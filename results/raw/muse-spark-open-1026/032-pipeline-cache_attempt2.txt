from cache import ResultCache


def _to_hashable(value):
    # Recursively convert mutable containers to immutable, hashable equivalents
    if isinstance(value, list):
        return tuple(_to_hashable(v) for v in value)
    if isinstance(value, tuple):
        return tuple(_to_hashable(v) for v in value)
    if isinstance(value, dict):
        # sort by hashable key representation to be deterministic
        return tuple(sorted((_to_hashable(k), _to_hashable(v)) for k, v in value.items()))
    if isinstance(value, set):
        return frozenset(_to_hashable(v) for v in value)
    # hashable leaf types (str, int, float, bool, None, etc.)
    try:
        hash(value)
        return value
    except TypeError:
        # fallback for unhashable custom objects: use repr which captures value
        return repr(value)


class Pipeline:
    def __init__(self, transforms):
        self._transforms = transforms
        self._cache = ResultCache()

    def run(self, data):
        result = data
        for t in self._transforms:
            # cache key captures full contents of the input, not identity
            key = (t.name, _to_hashable(result))
            if self._cache.has(key):
                result = self._cache.get(key)
            else:
                result = t.apply(result)
                self._cache.set(key, result)
        return result
