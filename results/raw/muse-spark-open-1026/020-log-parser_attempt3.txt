import datetime


def duration_seconds(entries):
    """
    Return elapsed seconds between first and last entry.
    Returns 0 if fewer than 2 entries.
    Correctly handles logs that span midnight by tracking day rollover.
    """
    if len(entries) < 2:
        return 0
    base = datetime.date(2000, 1, 1)
    first = datetime.datetime.combine(base, entries[0].time)
    prev = first
    for entry in entries[1:]:
        cur = datetime.datetime.combine(prev.date(), entry.time)
        # if time went backwards, we have crossed midnight -> next day
        if cur < prev:
            cur += datetime.timedelta(days=1)
            # handle unlikely case of multiple day gap in one step
            while cur < prev:
                cur += datetime.timedelta(days=1)
        prev = cur
    last = prev
    delta = last - first
    return delta.total_seconds()
