import re


def slugify(text):
    """Convert text to a URL slug."""
    # Replace non-alphanumeric characters with hyphens
    slug = re.sub(r"[^a-zA-Z0-9]+", "-", text)
    # lowercase the result
    slug = slug.lower()
    # collapse consecutive hyphens to one
    slug = re.sub(r"-+", "-", slug)
    # strip leading/trailing hyphens
    slug = slug.strip("-")
    return slug
