import datetime


def duration_seconds(entries):
    """
    Return elapsed seconds between first and last entry.
    Returns 0 if fewer than 2 entries.
    Handles logs that span midnight by detecting time going backwards
    and advancing the date accordingly.
    """
    if len(entries) < 2:
        return 0
    base = datetime.date(2000, 1, 1)
    first = datetime.datetime.combine(base, entries[0].time)
    prev = first
    current_date = base
    # Walk through entries to account for midnight crossing(s)
    for entry in entries[1:]:
        cur = datetime.datetime.combine(current_date, entry.time)
        if cur < prev:
            current_date += datetime.timedelta(days=1)
            cur = datetime.datetime.combine(current_date, entry.time)
        prev = cur
    last = prev
    delta = last - first
    return delta.total_seconds()
