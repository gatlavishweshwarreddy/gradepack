import datetime


def duration_seconds(entries):
    """
    Return elapsed seconds between first and last entry.
    Returns 0 if fewer than 2 entries.
    Handles logs that span midnight (and multiple midnights) by
    detecting when the clock wraps around.
    """
    if len(entries) < 2:
        return 0
    base = datetime.date(2000, 1, 1)
    first = datetime.datetime.combine(base, entries[0].time)
    prev_time = entries[0].time
    day_offset = 0
    last = first
    for entry in entries[1:]:
        cur_time = entry.time
        # clock went backwards -> crossed midnight
        if cur_time < prev_time:
            day_offset += 1
        last = datetime.datetime.combine(
            base + datetime.timedelta(days=day_offset), cur_time
        )
        prev_time = cur_time
    delta = last - first
    return delta.total_seconds()
