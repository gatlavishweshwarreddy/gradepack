from cache import ResultCache


def _freeze(obj):
    # Recursively convert mutable containers into immutable, hashable equivalents
    # so the cache key captures the full contents of the input.
    if isinstance(obj, list):
        return tuple(_freeze(x) for x in obj)
    if isinstance(obj, tuple):
        return tuple(_freeze(x) for x in obj)
    if isinstance(obj, dict):
        return tuple(sorted((_freeze(k), _freeze(v)) for k, v in obj.items()))
    if isinstance(obj, set):
        return frozenset(_freeze(x) for x in obj)
    # For other types, try to use the object directly if hashable, otherwise fallback to repr
    try:
        hash(obj)
        return obj
    except TypeError:
        return repr(obj)


class Pipeline:
    def __init__(self, transforms):
        self._transforms = transforms
        self._cache = ResultCache()

    def run(self, data):
        result = data
        for t in self._transforms:
            key = (t.name, _freeze(result))
            if self._cache.has(key):
                result = self._cache.get(key)
            else:
                result = t.apply(result)
                self._cache.set(key, result)
        return result
