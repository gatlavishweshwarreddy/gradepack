def flatten(obj, sep="."):
    """Flatten a nested dict to dot-separated keys."""
    result = {}

    def _recurse(current, prefix):
        for key, value in current.items():
            if prefix:
                new_key = prefix + sep + key
            else:
                new_key = key
            if isinstance(value, dict):
                _recurse(value, new_key)
            else:
                result[new_key] = value

    _recurse(obj, "")
    return result
