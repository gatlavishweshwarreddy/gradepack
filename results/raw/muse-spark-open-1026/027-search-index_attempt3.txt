class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._built = False
        self.comparison_count = 0
        self._index = {}

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        self._index = {}
        for doc_id, text in self._docs.items():
            n = len(text)
            # avoid duplicate substrings within the same document
            seen = set()
            for i in range(n):
                # slicing creates substring text[i:j]
                for j in range(i + 1, n + 1):
                    substr = text[i:j]
                    if substr in seen:
                        continue
                    seen.add(substr)
                    if substr not in self._index:
                        self._index[substr] = [doc_id]
                    else:
                        self._index[substr].append(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query using inverted index."""
        q = query.lower()
        if q == "":
            return list(self._docs.keys())
        if not self._built:
            self.build()
        # O(1) dictionary lookup, no per-document string comparison
        res = self._index.get(q)
        if res is None:
            return []
        return list(res)
