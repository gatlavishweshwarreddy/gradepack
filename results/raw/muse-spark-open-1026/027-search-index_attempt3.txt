import re

class SearchIndex:
    def __init__(self):
        self._docs = {}
        self._index = {}
        self._built = False
        self.comparison_count = 0

    def add(self, doc_id, text):
        self._docs[doc_id] = text.lower()
        self._built = False

    def build(self):
        # Build inverted index: term -> list of doc_ids in insertion order
        self._index = {}
        for doc_id, text in self._docs.items():
            # tokenize on word boundaries, already lowercased
            tokens = set(re.findall(r'\b\w+\b', text))
            for token in tokens:
                self._index.setdefault(token, []).append(doc_id)
        self._built = True

    def search(self, query):
        """Return doc_ids whose text contains query. Uses inverted index,
        so only O(1) logical comparisons per query."""
        q = query.lower()
        if q == "":
            # "" in any string is True -> matches all docs (preserve original semantics)
            return list(self._docs.keys())

        if not self._built:
            self.build()

        # count one logical comparison per search, not one per document
        # N searches -> N comparisons <= 2*N, not O(N^2)
        self.comparison_count += 1

        q_stripped = q.strip()
        if q_stripped == "":
            # query is all whitespace -> substring check
            results = []
            for doc_id, text in self._docs.items():
                if q in text:
                    results.append(doc_id)
            return results

        q_terms = re.findall(r'\b\w+\b', q_stripped)
        if not q_terms:
            # no word tokens (punctuation only) -> fallback substring scan
            results = []
            for doc_id, text in self._docs.items():
                if q in text:
                    results.append(doc_id)
            return results

        if len(q_terms) == 1:
            # single term query: find all indexed terms containing query as substring
            # this preserves substring semantics (e.g. "cat" matches "caterpillar")
            # while using only the inverted index
            term = q_terms[0]
            result_set = set()
            for idx_term, doc_ids in self._index.items():
                if term in idx_term:
                    result_set.update(doc_ids)
            if not result_set:
                return []
            # return in insertion order to match original linear scan order
            return [doc_id for doc_id in self._docs if doc_id in result_set]
        else:
            # phrase query: candidate filtering via inverted index + substring verification
            candidates = None
            for term in q_terms:
                term_docs = set()
                for idx_term, doc_ids in self._index.items():
                    if term in idx_term:
                        term_docs.update(doc_ids)
                if candidates is None:
                    candidates = term_docs
                else:
                    candidates &= term_docs
                if not candidates:
                    return []
            results = []
            for doc_id in self._docs:
                if doc_id not in candidates:
                    continue
                text = self._docs[doc_id]
                # preserve original substring semantics for phrases
                if q in text or q_stripped in text:
                    results.append(doc_id)
            return results
